# 7.1 搭建一层输入∕输出安全护栏——Llama Guard 安全防护模型实战

> **硬件要求**：示例面向单张 CUDA GPU、batch size=1、短文本。Llama Guard 1B 与 Qwen 1.5B 半精度权重合计约 5GB，另需运行时、激活与 KV cache；建议 8–12GB 以上显存，实际以长度和实测峰值为准，不能把 8GB 当保证。CPU/MPS、量化与多卡需另行适配，本节不声称已验证。
>
> **本节目标**：实际接入 Llama Guard，组装 **输入审核 → 生成 → 带上下文的输出审核 → 返回**。严格区分 `safe`、`unsafe` 和 `error`，异常默认不放行；用无害样例与模拟审核结果测试控制流，不提供攻击操作指南。

## 一、专业安全模型：输出标签先解析，再决定是否放行

`meta-llama/Llama-Guard-3-1B` 是内容分类模型，不是权限系统。它根据专用模板评估对话中的最后一条 `user` 或 `assistant` 消息。下面保留默认政策与类别，不自定义类别；若更换模型或政策，解析约定也要同步变更。

模型需要在 Hugging Face 接受许可并获得访问权限；访问失败应停止实验，不能自动切换为无审核生成。模型卡声明支持的八种语言**不包括中文**：英文样例用于演示官方支持范围内的调用，中文样例仅做域外观察，不代表中文安全质量得到验证。

```python
import re
import torch
from transformers import AutoTokenizer, AutoModelForCausalLM

if not torch.cuda.is_available():
    raise RuntimeError("此示例按单张 CUDA GPU 编写；CPU/MPS 需另行调整。")
dtype = torch.bfloat16 if torch.cuda.is_bf16_supported() else torch.float16
guard_name = "meta-llama/Llama-Guard-3-1B"
guard_tok = AutoTokenizer.from_pretrained(guard_name)
guard_model = AutoModelForCausalLM.from_pretrained(
    guard_name, torch_dtype=dtype, device_map={"": 0}
).eval()


def parse_guard(text):
    text = text.strip()
    if text == "safe":
        return {"status": "safe", "categories": []}
    match = re.fullmatch(r"unsafe\s*\n\s*(S\d+(?:\s*,\s*S\d+)*)", text)
    if match:
        categories = [x.strip() for x in match.group(1).split(",")]
        if all(x in {f"S{i}" for i in range(1, 14)} for x in categories):
            return {"status": "unsafe", "categories": categories}
    return {"status": "error", "reason": "invalid_guard_output"}


def ended_with_eos(ids, model, tok):
    eos = model.generation_config.eos_token_id
    eos_ids = {eos} if isinstance(eos, int) else set(eos or [tok.eos_token_id])
    return len(ids) > 0 and ids[-1].item() in eos_ids


@torch.inference_mode()
def check_safety(messages):
    try:
        if not messages or any(
            m.get("role") not in {"user", "assistant"}
            or not isinstance(m.get("content"), str) or not m["content"].strip()
            for m in messages
        ):
            return {"status": "error", "reason": "invalid_messages"}
        conversation = [
            {"role": m["role"], "content": [{"type": "text", "text": m["content"]}]}
            for m in messages
        ]
        input_ids = guard_tok.apply_chat_template(
            conversation, return_tensors="pt"
        ).to(guard_model.device)
        limit = min(4096, guard_model.config.max_position_embeddings - 64)
        if input_ids.shape[1] > limit:
            return {"status": "error", "reason": "guard_context_too_long"}
        output = guard_model.generate(
            input_ids=input_ids, attention_mask=torch.ones_like(input_ids),
            max_new_tokens=64, do_sample=False, pad_token_id=guard_tok.eos_token_id,
        )
        new_ids = output[0, input_ids.shape[1]:]
        if not ended_with_eos(new_ids, guard_model, guard_tok):
            return {"status": "error", "reason": "guard_not_finished"}
        return parse_guard(guard_tok.decode(new_ids, skip_special_tokens=True))
    except Exception:
        return {"status": "error", "reason": "guard_unavailable"}


test_prompts = [
    "Explain the basic idea of a Transformer.",
    "How can I protect my account with multi-factor authentication?",
    "介绍一下 Transformer 的原理。",
]
for prompt in test_prompts:
    print(check_safety([{"role": "user", "content": prompt}]))
```

应用层只接受完整的 `safe`，或 `unsafe` 加一行合法类别编号。空输出、夹带解释、未知编号、截断与运行异常都返回 `error`，**不是安全**。这比 `"unsafe" in result` 更保守，但仍只是语法校验，不能证明模型分类正确。模型升级后格式不兼容可能增加拦截，应通过离线测试处理，而不是放松成任意子串匹配。

## 二、生成候选回答：安全提示不是权限边界

通用模型可以辅助审查，但将用户输入直接拼接进“请判断 safe/unsafe”的提示，可能让待审核文本干扰审核指令。它也容易与生成器犯相关错误，因此本节**不将自我审查当作 Llama Guard 不可用时的安全替代**。专用分类器同样可能被干扰，仍需外部权限与离线评测。

```python
chat_name = "Qwen/Qwen2.5-1.5B-Instruct"
chat_tok = AutoTokenizer.from_pretrained(chat_name)
chat_model = AutoModelForCausalLM.from_pretrained(
    chat_name, torch_dtype=dtype, device_map={"": 0}
).eval()
safety_system = (
    "你是乐于助人的助手。尊重隐私，避免有害内容；无法安全满足的请求应说明边界，"
    "并提供安全的替代帮助。对普通学习和防护问题正常回答。"
)


@torch.inference_mode()
def generate_reply(user_text):
    messages = [
        {"role": "system", "content": safety_system},
        {"role": "user", "content": user_text},
    ]
    text = chat_tok.apply_chat_template(messages, tokenize=False, add_generation_prompt=True)
    inputs = chat_tok(text, return_tensors="pt", add_special_tokens=False).to(chat_model.device)
    if inputs["input_ids"].shape[1] > min(2048, chat_model.config.max_position_embeddings - 512):
        raise ValueError("generation_context_too_long")
    output = chat_model.generate(
        **inputs, max_new_tokens=512, do_sample=False,
        pad_token_id=chat_tok.eos_token_id,
    )
    new_ids = output[0, inputs["input_ids"].shape[1]:]
    if not ended_with_eos(new_ids, chat_model, chat_tok):
        raise RuntimeError("generation_not_finished")
    answer = chat_tok.decode(new_ids, skip_special_tokens=True).strip()
    if not answer:
        raise RuntimeError("empty_generation")
    return answer
```

此函数只生成**候选**回答，不应由服务接口直接返回；下一节把它放在两次审核之间。生成限长、空结果和异常都会阻止候选流出。这里不做流式输出，也不调用外部工具。

## 三、组装输入与输出两道审核

```python
def safe_chat(user_text, moderator_fn=check_safety, generator_fn=generate_reply):
    if not isinstance(user_text, str) or not user_text.strip():
        return {"status": "error", "stage": "input", "answer": "输入不能为空。"}
    stage = "input"
    try:
        conversation = [{"role": "user", "content": user_text}]
        verdict = moderator_fn(conversation)
        if verdict.get("status") != "safe":
            status = "blocked" if verdict.get("status") == "unsafe" else "error"
            return {"status": status, "stage": stage, "answer": "请求未通过审核或审核暂不可用。"}
        stage = "generation"
        candidate = generator_fn(user_text)
        if not isinstance(candidate, str) or not candidate.strip():
            raise ValueError("empty_generation")
        stage = "output"
        verdict = moderator_fn(conversation + [{"role": "assistant", "content": candidate}])
        if verdict.get("status") != "safe":
            status = "blocked" if verdict.get("status") == "unsafe" else "error"
            return {"status": status, "stage": stage, "answer": "候选回答未通过审核或审核暂不可用。"}
        return {"status": "ok", "stage": "done", "answer": candidate}
    except Exception:
        return {"status": "error", "stage": stage, "answer": "当前无法安全完成请求，请稍后再试。"}


for prompt in test_prompts:
    result = safe_chat(prompt)
    print(result["status"], result["stage"], result["answer"])
```

输出审核实际传入 `[user, assistant]`，而不是孤立地审核回答或把回答改成新的用户提问。模板根据最后一条角色识别审核对象。多轮服务必须同时保留真实用户/助手上下文，不能仅截取最后一句；本例限定为单轮文本。System Prompt 只用于生成，审核模型使用自身专用政策模板，不把生成器指令冒充对话内容。

任何阶段失败都不返回候选原文；线上也不要将候选原文直接写进普通日志。这里采用保守的 fail-closed 策略，会增加误拦截与不可用率，需结合重试、人工升级、脱敏审计与服务超时机制设计，不能因此忽略可用性。

## 四、先测试控制流，再评估分类质量

以下测试只模拟审核器状态，候选内容始终无害，用来确认“是否调用生成”和“是否泄露未审核候选”。它们不衡量 Llama Guard 的分类准确率。

```python
assert parse_guard("safe")["status"] == "safe"
assert parse_guard("unsafe\nS7")["status"] == "unsafe"
for malformed in ["", "not unsafe", "safe because ...", "unsafe", "unsafe\nS99"]:
    assert parse_guard(malformed)["status"] == "error"


def check_flow(labels, expected_status, expected_stage, expected_calls):
    labels = iter(labels)
    calls = []

    def moderator(messages):
        calls.append(messages[-1]["role"])
        if messages[-1]["role"] == "assistant":
            assert messages[0]["content"] == "Explain a tensor."
            assert messages[-1]["content"] == "A tensor is a multidimensional array."
        return {"status": next(labels)}

    def generator(text):
        calls.append("generate")
        return "A tensor is a multidimensional array."

    result = safe_chat("Explain a tensor.", moderator, generator)
    assert (result["status"], result["stage"]) == (expected_status, expected_stage)
    assert calls == expected_calls
    if expected_status != "ok":
        assert result["answer"] != "A tensor is a multidimensional array."


check_flow(["unsafe"], "blocked", "input", ["user"])
check_flow(["error"], "error", "input", ["user"])
check_flow(["safe", "unsafe"], "blocked", "output", ["user", "generate", "assistant"])
check_flow(["safe", "error"], "error", "output", ["user", "generate", "assistant"])
check_flow(["safe", "safe"], "ok", "done", ["user", "generate", "assistant"])


def unavailable(_):
    raise RuntimeError("simulated_unavailable")


assert safe_chat("Explain a tensor.", unavailable)["status"] == "error"
assert safe_chat("Explain a tensor.", lambda _: {"status": "safe"}, unavailable)["stage"] == "generation"
```

接着应在获得授权、隔离且不执行工具的环境中建立内容政策测试集，分别统计正常请求误拦率、不安全内容漏放率、审核异常率和额外延迟。多语言、多轮、隐私、无害引用和长度边界都需要覆盖；不能用三条演示请求证明生产安全。内容为 `safe` 也不能授权读文件、执行命令或外发数据，必须另外检查身份、权限与工具参数。
