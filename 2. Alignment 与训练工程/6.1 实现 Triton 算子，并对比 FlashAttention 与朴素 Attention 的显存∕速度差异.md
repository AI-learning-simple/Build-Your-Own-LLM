## 二、亲手写一个最简单的 Triton 算子

Triton 是一个让你能用近似 Python 的语法编写 GPU Kernel 的工具，不需要直接写 CUDA C++。我们从最简单的例子开始：向量加法。

```python
import triton
import triton.language as tl

@triton.jit
def add_kernel(x_ptr, y_ptr, out_ptr, n_elements, BLOCK_SIZE: tl.constexpr):
    # 每个 GPU 线程块（Block）负责处理一段连续的数据，pid 就是当前 Block 的编号
    pid = tl.program_id(axis=0)
    # 计算当前 Block 负责处理的数据在整个数组中的起始位置
    block_start = pid * BLOCK_SIZE
    offsets = block_start + tl.arange(0, BLOCK_SIZE)
    # mask 防止越界：最后一个 Block 可能不满BLOCK_SIZE 个元素
    mask = offsets < n_elements

    x = tl.load(x_ptr + offsets, mask=mask)   # 从 HBM 读取一段 x
    y = tl.load(y_ptr + offsets, mask=mask)   # 从 HBM 读取一段 y
    output = x + y                             # 在计算核心上做加法（这一步几乎不耗时）
    tl.store(out_ptr + offsets, output, mask=mask)  # 把结果写回 HBM


def triton_add(x: torch.Tensor, y: torch.Tensor) -> torch.Tensor:
    output = torch.empty_like(x)
    n_elements = output.numel()
    BLOCK_SIZE = 1024
    # grid 决定启动多少个 Block，每个 Block 处理 BLOCK_SIZE 个元素
    grid = (triton.cdiv(n_elements, BLOCK_SIZE),)
    add_kernel[grid](x, y, output, n_elements, BLOCK_SIZE=BLOCK_SIZE)
    return output
```

```python
import time

size = 100_000_000  # 一亿个元素，足够大才能看出差异
x = torch.rand(size, device="cuda")
y = torch.rand(size, device="cuda")

# 先跑一次，触发 Triton 的 JIT 编译（第一次调用会包含编译耗时，不计入计时）
_ = triton_add(x, y)
torch.cuda.synchronize()

start = time.time()
out_triton = triton_add(x, y)
torch.cuda.synchronize()   # GPU 是异步执行的，必须同步之后才能拿到真实耗时
triton_time = time.time() - start

start = time.time()
out_torch = x + y          # PyTorch 原生实现，底层也是调用一个预先写好的 CUDA Kernel
torch.cuda.synchronize()
torch_time = time.time() - start

# 验证结果是否一致：手写算子必须和标准实现数值完全对齐，这是正确性的底线
print("结果是否一致:", torch.allclose(out_triton, out_torch))
print(f"Triton 算子耗时: {triton_time * 1000:.3f} ms")
print(f"PyTorch 原生耗时: {torch_time * 1000:.3f} ms")
```

这个例子本身不追求"比 PyTorch 更快"（PyTorch 的向量加法已经调用了高度优化的 CUDA Kernel），而是让你亲眼看到：**平时调用的每一个 `+`、`*`，背后都对应着一段像上面这样在成千上万个线程里并行执行的代码**——这正是第6章讲的"算子"最朴素的样子。

## 三、朴素 Attention 实现，并测量它的真实显存与耗时

```python
import torch
import torch.nn.functional as F

def naive_attention(q, k, v):
    # 第6章提到：朴素实现会产生一个完整的 N×N 中间矩阵，并把它写入/读出 HBM 多次
    d_k = q.shape[-1]
    scores = torch.matmul(q, k.transpose(-2, -1)) / (d_k ** 0.5)  # 第一次写入 HBM：N×N 的 scores
    attn = F.softmax(scores, dim=-1)                # 从 HBM 读出 scores，写回 HBM
    out = torch.matmul(attn, v)                                    # 从 HBM 读出 attn，计算最终输出
    return out


def measure(fn, q, k, v, name):
    torch.cuda.reset_peak_memory_stats()   # 清空显存峰值统计，保证每次测量互不干扰
    torch.cuda.synchronize()

    start = time.time()
    for _ in range(10):     # 多次运行取平均，减少单次测量的抖动
        out = fn(q, k, v)
    torch.cuda.synchronize()
    elapsed = (time.time() - start) / 10

    peak_mem = torch.cuda.max_memory_allocated() /1024 ** 2   # 转换成 MB，方便阅读
    print(f"[{name}] 平均耗时: {elapsed * 1000:.2f} ms  |峰值显存: {peak_mem:.1f} MB")
    return out, elapsed, peak_mem
```

```python
batch, heads, seq_len, d_k = 2, 8, 4096, 64# 序列长度设置得比较长，才能放大朴素实现的劣势

q = torch.randn(batch, heads, seq_len, d_k, device="cuda", dtype=torch.float16)
k = torch.randn(batch, heads, seq_len, d_k, device="cuda", dtype=torch.float16)
v = torch.randn(batch, heads, seq_len, d_k, device="cuda", dtype=torch.float16)

_, naive_time, naive_mem = measure(naive_attention, q, k, v, "朴素 Attention")
```

## 四、用 PyTorch 内置的 Flash Attention 对比同样的计算

PyTorch 2.0 之后，`F.scaled_dot_product_attention` 会自动选择底层最优的实现，其中就包括Flash Attention 后端。我们可以显式指定后端，来做一次公平对比：

```python
from torch.nn.attention import SDPBackend, sdpa_kernel

def flash_attention(q, k, v):
    # 强制使用 Flash Attention 后端，确保对比的是"同一个数学公式，不同的算子实现"
    with sdpa_kernel(SDPBackend.FLASH_ATTENTION):
        return F.scaled_dot_product_attention(q, k, v)


_, flash_time, flash_mem = measure(flash_attention, q, k, v, "Flash Attention")

# 再验证一次：两种实现的数学结果应该高度接近（浮点误差范围内），这是"算子优化不改变结果"的直接证明
naive_out = naive_attention(q, k, v)
flash_out = flash_attention(q, k, v)
print("两种实现结果是否接近:", torch.allclose(naive_out, flash_out, atol=1e-2))
```

## 五、结果对比

```python
print("\n===== 对比总结 =====")
print(f"{'实现方式':<15}{'平均耗时(ms)':<15}{'峰值显存(MB)':<15}")
print(f"{'朴素 Attention':<15}{naive_time*1000:<15.2f}{naive_mem:<15.1f}")
print(f"{'Flash Attention':<15}{flash_time*1000:<15.2f}{flash_mem:<15.1f}")
print(f"\n速度提升: {naive_time / flash_time:.2f}x")
print(f"显存节省: {(1 - flash_mem / naive_mem) * 100:.1f}%")
```

在序列长度 4096 这个规模下，你通常会观察到 Flash Attention 相比朴素实现有数倍的速度提升，同时峰值显存明显更低（序列越长，这个差距会越明显，因为朴素实现的 N×N 中间矩阵是随序列长度平方增长的，而 Flash Attention 从头到尾都没有真正生成这个完整矩阵）。这正是第6章讲的"算子级优化"在真实代码里的样子：**两种实现算出的是完全相同的数学结果，性能差距完全来自于对 HBM/SRAM 的使用方式不同。**
