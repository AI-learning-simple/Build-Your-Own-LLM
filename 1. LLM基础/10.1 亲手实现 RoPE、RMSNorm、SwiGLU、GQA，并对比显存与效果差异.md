## 二、亲手实现 RoPE，验证"相对位置"性质

```python
def build_rope_cache(seq_len, dim, base=10000):
    # 每一对维度用不同的频率，频率随维度增大而降低
    freqs = 1.0 / (base ** (torch.arange(0, dim, 2).float() / dim))
    positions = torch.arange(seq_len).float()
    #位置 × 频率，得到每个位置在每一对维度上应该旋转的角度
    angles = torch.outer(positions, freqs)   # shape: [seq_len, dim/2]
    return torch.cos(angles), torch.sin(angles)


def apply_rope(x, cos, sin):
    # x: [seq_len, dim]，把最后一维两两分组，当成"二维向量"来旋转
    x1, x2 = x[..., 0::2], x[..., 1::2]
    # 标准的二维旋转公式：x' = x*cos - y*sin，y' = x*sin + y*cos
    rotated_x1 = x1 * cos - x2 * sin
    rotated_x2 = x1 * sin + x2 * cos
    # 把旋转后的两组维度重新交错拼接回原来的形状
    out = torch.stack([rotated_x1, rotated_x2], dim=-1).flatten(-2)
    return out
```

```python
seq_len, dim = 8, 4
cos, sin = build_rope_cache(seq_len, dim)

# 构造两个内容完全相同的 Query/Key 向量，只是位置不同
q_at_pos2 = apply_rope(torch.tensor([[1.0, 0.5, -0.3, 0.8]]), cos[2:3], sin[2:3])
k_at_pos5 = apply_rope(torch.tensor([[1.0, 0.5, -0.3, 0.8]]), cos[5:6], sin[5:6])
q_at_pos102 = apply_rope(torch.tensor([[1.0, 0.5, -0.3, 0.8]]), cos[6:7], sin[6:7])  # 相对距离仍是3（6-3）
k_at_pos105 = apply_rope(torch.tensor([[1.0, 0.5, -0.3, 0.8]]), cos[3:4], sin[3:4])

# 验证：只要"相对距离"相同（都是3），点积结果应该高度接近，即使绝对位置完全不同
dot_2_5 = (q_at_pos2 * k_at_pos5).sum()
dot_6_3 = (q_at_pos102 * k_at_pos105).sum()
print(f"位置(2,5) 相对距离3的点积: {dot_2_5.item():.4f}")
print(f"位置(6,3) 相对距离3 的点积: {dot_6_3.item():.4f}")
print("两者是否接近（验证 RoPE 的相对位置性质）:", torch.allclose(dot_2_5, dot_6_3, atol=1e-4))
```

你会看到两次点积结果几乎完全一致——**这就是 RoPE 最核心的性质：Q、K 的点积只取决于它们的相对位置差，与绝对位置无关**，这也是第10章解释的"天然获得相对位置编码"的真实证据。

## 三、RMSNorm vs LayerNorm：速度对比

```python
class RMSNorm(nn.Module):
    def __init__(self, dim, eps=1e-6):
        super().__init__()
        self.weight = nn.Parameter(torch.ones(dim))   # 只保留缩放参数，没有偏置
        self.eps = eps

    def forward(self, x):
        rms = torch.sqrt(torch.mean(x ** 2, dim=-1, keepdim=True) + self.eps)
        return (x / rms) * self.weight


d_model = 4096
layernorm = nn.LayerNorm(d_model)
rmsnorm = RMSNorm(d_model)

x = torch.randn(64, 128, d_model)   # 模拟一个真实 Batch 的输入

# 分别计时 1000 次前向计算
for name, norm_fn in [("LayerNorm", layernorm), ("RMSNorm", rmsnorm)]:
    start = time.time()
    for _ in range(1000):
        _ = norm_fn(x)
    elapsed = time.time() - start
    print(f"{name}: {elapsed * 1000:.2f} ms / 1000次")
```

在 CPU 上你通常就能看到 RMSNorm 比 LayerNorm 略快一些（少了减均值和偏置项两步计算）；这个差距在千亿参数、每天训练消耗数万GPU 小时的真实场景下，会被放大成实实在在的时间和电费成本。

**那 BatchNorm 呢？** 在卷积网络时代，**批归一化（BatchNorm）**几乎是标配——它按"一个 Batch 内所有样本"统计均值方差。但 Transformer 出现后，NLP 模型普遍放弃了 BatchNorm，改用 LayerNorm（以及后来的 RMSNorm），原因有三：

1. **依赖 Batch**：BatchNorm 的统计量取决于同 Batch 里的其它样本——而 NLP 里序列长度可变、Batch 大小受显存限制往往很小，统计量不稳定；更麻烦的是推理时 Batch 可以只有 1 条，训练与推理行为不一致；
2. **小 Batch 噪声大**：LLM 训练显存紧张，Batch 往往很小，均值方差的估计噪声反而成了干扰；
3. **在 Transformer 上实测不好用**：LayerNorm 对单个样本自己归一化（不需要跨样本统计），行为稳定、与序列长度无关，这也是为什么从原始 Transformer 论文到今天的 Llama/Qwen，归一化选的一直是 LayerNorm / RMSNorm。

一句话总结：**BatchNorm 归一化"整个 Batch"，适合 CNN；LayerNorm/RMSNorm 归一化"单个样本自己"，适合序列模型。** 这也是本节对比时只出现 LayerNorm 和 RMSNorm 的原因——BatchNorm 在 LLM 的世界里几乎已经退场（Karpathy 的 makemore 课程里专门演示过 BatchNorm 的威力，但也正是在那里能看到它和序列任务的不匹配之处）。

## 四、SwiGLU vs GELU-MLP：参数量对比

```python
class GELUMLP(nn.Module):
    def __init__(self, d_model, d_ff):
        super().__init__()
        self.fc1 = nn.Linear(d_model, d_ff)
        self.fc2 = nn.Linear(d_ff, d_model)

    def forward(self, x):
        return self.fc2(F.gelu(self.fc1(x)))


class SwiGLU(nn.Module):
    def __init__(self, d_model, d_ff):
        super().__init__()
        self.gate_proj = nn.Linear(d_model, d_ff)
        self.up_proj = nn.Linear(d_model, d_ff)
        self.down_proj = nn.Linear(d_ff, d_model)

    def forward(self, x):
        return self.down_proj(F.silu(self.gate_proj(x)) * self.up_proj(x))


def count_params(model):
    return sum(p.numel() for p in model.parameters())

d_model = 768
gelu_mlp = GELUMLP(d_model, d_ff=3072)
# SwiGLU 多了一条门控分支，为了让总参数量接近，通常会把 d_ff 调小（这里用经验值 2/3）
swiglu = SwiGLU(d_model, d_ff=2048)

print(f"GELU-MLP 参数量: {count_params(gelu_mlp):,}")
print(f"SwiGLU 参数量:   {count_params(swiglu):,}")
```

可以看到，只要把 SwiGLU 的隐藏维度调小一些，两者的总参数量可以基本持平——这正是 Llama 等模型在设计 `d_ff` 时的实际做法：用差不多的参数预算，换来 SwiGLU 门控机制带来的效果提升。

## 五、GQA vs MHA：KV Cache 显存对比

```python
def kv_cache_size(num_heads, kv_groups, seq_len, head_dim, dtype_bytes=2):
    # kv_groups=num_heads 时就是 MHA（每个 Head 独立）
    # kv_groups=1 时就是 MQA（全局共享一份）
    # 1 < kv_groups < num_heads 时就是 GQA
    num_kv = kv_groups
    # 乘 2 是因为要同时存K 和 V
    total_bytes = 2 * num_kv * seq_len * head_dim * dtype_bytes
    return total_bytes / 1024 ** 2   # 转换成 MB


num_heads, head_dim, seq_len = 32, 128, 4096

configs = [
    ("MHA（32个独立KV）", 32),
    ("GQA（分8组）", 8),
    ("GQA（分4组）", 4),
    ("MQA（全局共享1份）", 1),
]

print(f"{'配置':<20}{'KV Cache大小(MB)':<20}")
for name, groups in configs:
    size_mb = kv_cache_size(num_heads, groups, seq_len, head_dim)
    print(f"{name:<20}{size_mb:<20.1f}")
```

在序列长度 4096、32 个 Head 的设定下，你会看到 GQA 分成 8 组之后，KV Cache 直接缩小到 MHA 的 1/4；分成 4 组则缩小到 1/8——这正是第10章说的"几乎免费的午餐"：**架构上一个小改动，换来推理阶段显存占用的大幅下降，这部分省下来的显存，恰好可以用来支撑更长的上下文或者更大的并发 Batch。**
