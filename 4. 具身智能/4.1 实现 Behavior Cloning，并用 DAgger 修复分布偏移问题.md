# 4.1 实现 Behavior Cloning，并用 DAgger 缓解分布偏移

> **本节定位**：实践。BC、状态分布偏移与 DAgger 原理见第4章。
>
> **硬件要求**：CPU + NumPy，无需机器人。依次运行各代码块。
>
> **实验目标**：手写两层策略及正确的 MSE 梯度；在固定部署起点分布上比较纯 BC 与 DAgger；用未参与聚合的起点评测，区分“扩展初始数据覆盖”与“聚合策略访问状态”。实验是经典算法的教学演示，不保证三轮即可解决所有偏移。

---

# 一、任务与专家：导航到原点，接近终点必须停止

动作是二维有界方向/速度比例指令，环境将每步位移系数限制为 `min(STEP, ||p||)`，再乘动作a。远处专家朝原点走，近处进一步减速以免跨过原点后反复折返；到达容差内立即终止。仅把单位方向乘固定步长，会在原点两侧形成二周期，不能误判成学习失败。

```python
import numpy as np

STEP, TOL, HORIZON = 0.3, 0.1, 40
rng = np.random.default_rng(42)

def expert_action(p):
    p = np.atleast_2d(p)
    distance = np.linalg.norm(p, axis=1, keepdims=True)
    return -p / np.maximum(distance, STEP)  # STEP*a 不会越过目标

def advance(p, a):
    # 限速度，且位移不超过当前到原点的距离；方向仍由策略决定。
    a = np.asarray(a, dtype=float)
    a = a / max(1.0, np.linalg.norm(a))
    return p + min(STEP, np.linalg.norm(p)) * a

def sample_starts(generator, n):
    radius = generator.uniform(1.0, 3.0, n)
    angle = generator.uniform(0, 2 * np.pi, n)
    return np.stack([radius * np.cos(angle), radius * np.sin(angle)], axis=1)

def expert_dataset(starts):
    states, labels = [], []
    for start in starts:
        p = start.copy()
        for _ in range(HORIZON):
            if np.linalg.norm(p) <= TOL:
                break
            a = expert_action(p)[0]
            states.append(p.copy())
            labels.append(a)
            p = advance(p, a)
    return np.array(states), np.array(labels)

# 从一开始就固定部署分布：所有象限、半径[1,3]。
# 初始示范、聚合rollout、最终留出测试分别使用独立样本。
demo_starts = sample_starts(rng, 8)
collect_starts = sample_starts(rng, 24)
test_starts = sample_starts(np.random.default_rng(2026), 100)
P_initial, A_initial = expert_dataset(demo_starts)
```

这避免先只从第一象限采示范、之后扩大到四象限，再把两项变化都算作 DAgger 的收益。若额外做“仅第一象限示范”的外推压力测试，应明确它还包含起点分布变化，不能单独归因于策略误差累积。

# 二、两层策略与手动反向传播

```python
def init_model(hidden=32, seed=0):
    g = np.random.default_rng(seed)
    return {"W1": g.normal(0, 0.3, (2, hidden)), "b1": np.zeros(hidden),
            "W2": g.normal(0, 0.3, (hidden, 2)), "b2": np.zeros(2)}

def forward(model, P):
    h = np.tanh(np.einsum("ni,ih->nh", P, model["W1"]) + model["b1"])
    return np.tanh(np.einsum("nh,ha->na", h, model["W2"]) + model["b2"]), h

def train_bc(model, P, A, epochs=2000, lr=0.05, verbose=True):
    for epoch in range(epochs):
        pred, h = forward(model, P)
        loss = np.mean((pred - A) ** 2)
        # mean 对 N*d 个元素平均，不是只除以样本数 N。
        d_pred = 2 * (pred - A) / pred.size
        d_z2 = d_pred * (1 - pred ** 2)
        d_z1 = np.einsum("na,ha->nh", d_z2, model["W2"]) * (1 - h ** 2)
        grads = {"W2": np.einsum("nh,na->ha", h, d_z2), "b2": d_z2.sum(axis=0),
                 "W1": np.einsum("ni,nh->ih", P, d_z1), "b1": d_z1.sum(axis=0)}
        for name, grad in grads.items():
            model[name] -= lr * grad
        if verbose and (epoch % 500 == 0 or epoch == epochs - 1):
            print(f"epoch {epoch:4d}  loss={loss:.6f}")
    return model

bc = train_bc(init_model(), P_initial, A_initial)
```

这里用 `einsum` 显式表达矩阵乘法（例如 `ni,ih->nh` 等价于 `P @ W1`），便于小规模CPU复现。第4章按样本平均的向量平方和与这里按所有元素平均的 MSE 相差动作维数 `d=2`；梯度必须与所打印的损失一致。`epoch==epochs-1` 使第1999轮确实会打印。预期是训练损失下降，但不预填未测的精确日志；小损失不代表闭环成功。

# 三、同一终止规则下的闭环评测

```python
def rollout(model, start):
    p = np.array(start, dtype=float)
    trajectory = [p.copy()]
    for _ in range(HORIZON):
        if np.linalg.norm(p) <= TOL:
            break
        action = forward(model, p[None])[0][0]
        p = advance(p, action)
        trajectory.append(p.copy())
    return np.array(trajectory)

def evaluate(model, starts):
    trajectories = [rollout(model, s) for s in starts]
    distances = np.array([np.linalg.norm(t[-1]) for t in trajectories])
    success = distances <= TOL
    lengths = np.array([len(t) - 1 for t in trajectories])
    return {"success_rate": float(success.mean()),
            "mean_final_distance": float(distances.mean()),
            "mean_steps": float(lengths.mean())}

print("BC 留出评测:", evaluate(bc, test_starts))
```

`test_starts` 不加入训练数据，也不用于选择轮数或调超参数；下面预先固定三轮后再评测。若反复依据这100个测试点调参，它们就成了验证集，需要另留最终测试集。

# 四、DAgger：只改变采集状态的来源

```python
dagger = {k: v.copy() for k, v in bc.items()}
P_train, A_train = P_initial.copy(), A_initial.copy()
for round_id in range(3):
    # trajectory[:-1] 是实际执行动作前的状态，不给终止点编造动作。
    visited = np.concatenate([rollout(dagger, s)[:-1] for s in collect_starts])
    P_train = np.concatenate([P_train, visited])
    A_train = np.concatenate([A_train, expert_action(visited)])
    dagger = train_bc(dagger, P_train, A_train, epochs=1000, verbose=False)
    print(f"DAgger 第{round_id + 1}轮，累计样本数={len(P_train)}")

# 控制训练预算：纯BC也额外更新相同的3000轮，不增加专家标签。
bc_equal_updates = {k: v.copy() for k, v in bc.items()}
train_bc(bc_equal_updates, P_initial, A_initial, epochs=3000, verbose=False)
print("同更新轮数 BC:", evaluate(bc_equal_updates, test_starts))
print("DAgger 留出评测:", evaluate(dagger, test_starts))
```

预期 DAgger 有机会降低策略自己造成的偏离、改善留出成功率；结果也可能接近，因为这个任务简单、初始 BC 已能泛化。不能写成“必然失败→彻底修复”。同更新轮数控制了优化次数，但 DAgger 的数据更多、每轮计算量更大、专家查询也更多，因此仍不是等算力或等标注预算比较。可再加一个**相同新增标签预算、采专家轨迹而非策略轨迹**的对照，以区分数据量收益和状态聚合收益。

# 五、思考与边界

1. 将初始示范缩减为2条，或加入固定分布的运动扰动，观察留出成功率，而不要临时扩大起点范围。
2. 在专家查询预算固定时，优先标注哪些高不确定或失败前的状态？
3. 若真实机器人已经掉出桌面或碰撞，专家未必能提供可恢复动作；DAgger 不代替安全控制、复位和质量检查。

# 本实验总结

- MSE 梯度与损失的归一化严格对应，终止和最大步数明确。
- BC 与 DAgger 面对同一部署起点分布，评测点不参与聚合。
- DAgger 仍是理解交互式数据聚合的重要基线；收益需要用闭环评测证明，而非“修复”一词保证。
