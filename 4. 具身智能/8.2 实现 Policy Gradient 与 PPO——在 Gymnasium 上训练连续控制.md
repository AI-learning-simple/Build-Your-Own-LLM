## 一、从 REINFORCE 到裁剪代理目标

REINFORCE 用 `∇log π(a|s) × G` 估计期望回报梯度；加入不依赖当前动作的 baseline 可降低方差。下面是 CartPole 的最小独立示例，优化的是每局最多 500 步的**有限回合总回报**，所以本段把 TimeLimit 也作为回合结束，不使用价值 bootstrap。这与后面无限时域价值近似的处理不同。

```python
import gymnasium as gym
import numpy as np
import torch
from torch import nn
from torch.distributions import Categorical, Normal

torch.set_num_threads(1)

def mlp(n_in, n_out):
    return nn.Sequential(nn.Linear(n_in, 64), nn.Tanh(),
                         nn.Linear(64, 64), nn.Tanh(), nn.Linear(64, n_out))

def reinforce_cartpole(seed=0, episodes=100, lr=1e-3):
    torch.manual_seed(seed)
    env = gym.make("CartPole-v1")
    actor = mlp(4, 2)
    opt = torch.optim.Adam(actor.parameters(), lr=lr)
    baseline = 0.0
    scores = []
    try:
        for ep in range(episodes):
            obs, _ = env.reset(seed=seed * 10000 + ep)
            log_probs, score = [], 0.0
            while True:
                dist = Categorical(logits=actor(torch.as_tensor(obs, dtype=torch.float32)))
                action = dist.sample()
                log_probs.append(dist.log_prob(action))
                obs, reward, terminated, truncated, _ = env.step(int(action.item()))
                score += reward
                if terminated or truncated:
                    break
            # 使用之前各回合的 baseline，不用本回合动作更新后的 baseline。
            loss = -torch.stack(log_probs).sum() * (score - baseline)
            opt.zero_grad()
            loss.backward()
            opt.step()
            baseline = 0.9 * baseline + 0.1 * score
            scores.append(score)
    finally:
        env.close()
    return actor, scores
```

单回合回报、高学习率、未归一化的轨迹长度都会影响 REINFORCE 的方差与更新尺度，出现回退并不能归因于“缺少 clip”这一个因素。与下文 PPO 相比，本段还改变了 critic、学习率、batch、回报定义、优势估计和每批更新轮数，**不能当作 clip 的因果对照**。

PPO 使用 `ratio = exp(logπ_new − logπ_old)` 与固定优势 `A`：

```math
L^{clip}=\mathbb{E}[\min(ratio\,A,\operatorname{clip}(ratio,1-\epsilon,1+\epsilon)A)]
```

- `A > 0` 且 `ratio > 1+ε` 时，该样本继续提高概率的激励被裁掉；概率过低一侧仍可有梯度。
- `A < 0` 且 `ratio < 1−ε` 时，该样本继续降低概率的激励被裁掉；概率过高一侧仍可有梯度。
- 这不是把实际 ratio、参数步长或 KL 硬限制在区间内。参数共享、其他样本以及多轮优化仍可能让策略大幅变化，clip 不提供“单步破坏力封顶”或“训练不崩”的保证。

## 二、一个实现，同时支持 CartPole 与 Pendulum

CartPole 是 4 维连续观测、2 个离散动作；Pendulum 是 **3 维观测 `[cosθ, sinθ, θ_dot]`、1 维动作 ndarray，范围 `[-2, 2]`**。连续策略先采样潜变量 `u`，再执行 `a=2 tanh(u)`，避免“执行了 clipped action，却错误地给 clipped 值套上原始高斯 log_prob”。下面显式保留 `u` 并计算变换后的 log_prob；雅可比项在同一动作的新旧比值中抵消，但仍完整写出。

本段将两个环境的 TimeLimit 视为外部采样截断，近似优化折扣的继续任务：`terminated` 不 bootstrap，`truncated` 和 rollout 最后一帧要 bootstrap。GAE 不跨 reset 传播。**若任务定义本身就是有限时域，需把剩余时间纳入状态，并在真正时间终点令末 V=0，不能机械套用这里的语义。**

```python
class ActorCritic(nn.Module):
    def __init__(self, obs_dim, act_dim, continuous):
        super().__init__()
        self.continuous = continuous
        self.actor = mlp(obs_dim, act_dim)
        self.critic = mlp(obs_dim, 1)
        if continuous:
            self.log_std = nn.Parameter(torch.zeros(act_dim))

    def distribution(self, obs):
        out = self.actor(obs)
        if self.continuous:
            return Normal(out, self.log_std.clamp(-5, 2).exp())
        return Categorical(logits=out)

    def log_prob(self, obs, latent_action):
        dist = self.distribution(obs)
        if not self.continuous:
            return dist.log_prob(latent_action)
        u = latent_action
        # log(1-tanh(u)^2) 的数值稳定形式，外加缩放系数 2 的雅可比。
        log_jac = np.log(2.0) + 2 * (np.log(2.0) - u - nn.functional.softplus(-2 * u))
        return (dist.log_prob(u) - log_jac).sum(-1)

    def value(self, obs):
        return self.critic(obs).squeeze(-1)

    @torch.no_grad()
    def act(self, obs, deterministic=False):
        dist = self.distribution(obs)
        if self.continuous:
            latent = dist.mean if deterministic else dist.sample()
            action = (2.0 * torch.tanh(latent)).cpu().numpy().astype(np.float32)
        else:
            latent = dist.logits.argmax(-1) if deterministic else dist.sample()
            action = int(latent.item())
        return action, latent, self.log_prob(obs, latent), self.value(obs)


def train(env_id="Pendulum-v1", seed=0, use_clip=True,
          updates=40, rollout_steps=1024, epochs=4, lr=3e-4,
          gamma=0.99, gae_lambda=0.95, clip_eps=0.2):
    if env_id not in ("CartPole-v1", "Pendulum-v1"):
        raise ValueError("本例只适配 CartPole-v1 / Pendulum-v1")
    torch.manual_seed(seed)
    np.random.seed(seed)
    env = gym.make(env_id)
    continuous = env_id == "Pendulum-v1"
    ac = ActorCritic(env.observation_space.shape[0], 1 if continuous else 2, continuous)
    opt = torch.optim.Adam(ac.parameters(), lr=lr)
    obs, _ = env.reset(seed=seed)
    scores, episode_score = [], 0.0
    try:
        for update in range(updates):
            states, actions, old_lps, values, rewards, next_values, ends = ([] for _ in range(7))
            for _ in range(rollout_steps):
                s = torch.as_tensor(obs, dtype=torch.float32)
                action, latent, lp, value = ac.act(s)
                next_obs, reward, terminated, truncated, _ = env.step(action)
                with torch.no_grad():
                    # next_obs 是 reset 之前的末观测，不是下一回合初始观测。
                    nv = 0.0 if terminated else ac.value(torch.as_tensor(next_obs, dtype=torch.float32)).item()
                states.append(s)
                actions.append(latent)
                old_lps.append(lp)
                values.append(value.item())
                rewards.append(float(reward))
                next_values.append(nv)
                ends.append(terminated or truncated)
                episode_score += reward
                if terminated or truncated:
                    scores.append(episode_score)
                    episode_score = 0.0
                    obs, _ = env.reset()
                else:
                    obs = next_obs

            adv = np.zeros(rollout_steps, dtype=np.float32)
            gae = 0.0
            for t in reversed(range(rollout_steps)):
                delta = rewards[t] + gamma * next_values[t] - values[t]
                gae = delta + gamma * gae_lambda * (not ends[t]) * gae
                adv[t] = gae
            targets = torch.tensor(adv + np.asarray(values, dtype=np.float32))
            advantages = torch.tensor(adv)
            advantages = (advantages - advantages.mean()) / (advantages.std(unbiased=False) + 1e-8)
            states = torch.stack(states)
            actions = torch.stack(actions)
            old_lps = torch.stack(old_lps).detach()
            # 旧 log_prob、优势和 target 在全部 epoch 内固定。
            for _ in range(epochs):
                new_lps = ac.log_prob(states, actions)
                ratio = (new_lps - old_lps).exp()
                surrogate = ratio * advantages
                if use_clip:  # 对照组唯一主动改变的开关
                    surrogate = torch.minimum(surrogate, ratio.clamp(1-clip_eps, 1+clip_eps) * advantages)
                loss = -surrogate.mean() + 0.5 * (ac.value(states) - targets).square().mean()
                if not torch.isfinite(loss):
                    raise RuntimeError("非有限 loss：停止训练并检查数值/超参数")
                opt.zero_grad()
                loss.backward()
                nn.utils.clip_grad_norm_(ac.parameters(), 0.5)  # 两组共有，不是 PPO clip
                opt.step()
            with torch.no_grad():
                log_ratio = ac.log_prob(states, actions) - old_lps
                approx_kl = ((log_ratio.exp() - 1) - log_ratio).mean().item()
            if (update + 1) % 10 == 0:
                avg = np.mean(scores[-10:]) if scores else float("nan")
                print(env_id, seed, use_clip, (update+1)*rollout_steps, avg, approx_kl)
    finally:
        env.close()
    return ac, scores


def evaluate(ac, env_id, seeds=range(10000, 10020)):
    env = gym.make(env_id)
    returns = []
    try:
        for seed in seeds:
            obs, _ = env.reset(seed=seed)
            score = 0.0
            while True:
                action, _, _, _ = ac.act(torch.as_tensor(obs, dtype=torch.float32), deterministic=True)
                obs, reward, terminated, truncated, _ = env.step(action)
                score += reward
                if terminated or truncated:
                    break
            returns.append(score)
    finally:
        env.close()
    return np.asarray(returns)

# 默认只做一次完整 Pendulum 训练；小机器先把 updates 改成 1 检查接口。
if __name__ == "__main__":
    policy, train_scores = train()
    evaluation = evaluate(policy, "Pendulum-v1")
    print("独立评测均值/标准差:", evaluation.mean(), evaluation.std())
```

代码按顺序放入同一个 Python 文件即可；本节没有安装/训练实测，因此不提供伪造的收敛表。Pendulum 回报通常为负，越接近 0 越好；CartPole 每回合评测上限是 500，训练曲线不等于独立评测结果。

## 三、怎样做有效对照

在同一环境上分别运行 `train(seed=s, use_clip=False)` 和 `train(seed=s, use_clip=True)`，其余参数全部相同。两者都是相同的 Actor-Critic / GAE / 多 epoch 代理目标实现，区别只有 clip 开关；**无 clip 组不是上面的单轨迹 REINFORCE**。固定 `updates × rollout_steps`，而不是固定局数，否则每局变长会偷偷改变样本预算。

建议至少使用 5 个训练种子（如 `0..4`），各组使用同一组独立评测种子 `10000..10019`，记录每个训练种子的评测均值，再汇总跨种子的均值/标准差或区间。记录软件版本、超参数、训练步数、耗时、approx-KL 和非有限值事件。相同随机种子不保证两种策略之后访问相同状态；这是控制初始条件，而非强行让轨迹相同。

`clip_eps=1.0` 不等于“关闭裁剪”：它仍然在特定方向限制代理目标。直接用布尔开关才能明确对照。若要加入 KL early stopping，两组都应采用同一规则并记录实际优化步数，否则又引入额外变量。

## 四、思考与总结

- 表格离散化可以近似连续空间，但维度与精度提升会使表规模迅速增长；神经网络不是唯一选择，策略梯度也不是连续控制的唯一算法。
- Critic、GAE、batch、学习率、梯度范数裁剪与 PPO clip 是不同机制，不能把一张训练曲线的变化全部归功于 clip。
- `terminated`、`truncated`、rollout 边界和真正有限时间目标需要分别处理。
- 本例保留经典 PPO 的教学价值，但不代表完整工程实现，也不能直接驱动真实机器人。RLHF 与机器人 PPO 共享代理目标思想，动作分布、奖励、KL 正则、采样和实现细节仍有差异。
