# 4.2 实现逆强化学习与 GAIL——从专家示范中学会奖励

> **本节定位**：经典算法实践，理论定位与局限见第4章第五节。
>
> **硬件要求**：CPU、Python 3.9+、NumPy。依次执行代码块，无需深度学习框架。
>
> **实验目标**：在同一个有限时域网格中实现最大熵 IRL 与表格 GAIL，重点验证终止、奖励归属、占用统计和策略梯度的一致性，而不是宣称恢复了唯一“真实奖励”。

# 一、共用环境：有限时域，到达目标立即结束

4×4网格，最多 `H=12` 次转移；本节用有限时域**未折扣**目标（`γ=1`），不混入无限时域折扣价值迭代。达到 `GOAL` 后无动作、无奖励、无熵；到时限同样结束。轨迹保存 `(s,a,s_next)`，终点只作为最后一次转移的 `s_next`，不虚构“终点向上走”的动作。

```python
import numpy as np

rng = np.random.default_rng(42)
N, A, H = 4, 4, 12
S, GOAL = N * N, N * N - 1
MOVES = [(-1, 0), (1, 0), (0, -1), (0, 1)]

def step(s, a):
    if s == GOAL:
        return GOAL  # 表内吸收，但采样/规划均不从终点选动作
    r, c = divmod(s, N)
    dr, dc = MOVES[a]
    return int(np.clip(r + dr, 0, N - 1) * N + np.clip(c + dc, 0, N - 1))

NEXT = np.array([[step(s, a) for a in range(A)] for s in range(S)])
STARTS = np.arange(GOAL)  # 多起点，而不是反复复制同一条起点轨迹
p0 = np.zeros(S)
p0[STARTS] = 1 / len(STARTS)

def sample_traj(policy, start):
    s, trajectory = int(start), []
    for t in range(H):
        if s == GOAL:
            break
        a = int(policy(t, s))
        ns = int(NEXT[s, a])
        trajectory.append((s, a, ns))
        s = ns
    return trajectory

def expert(t, s):
    r, c = divmod(s, N)
    return 1 if r < N - 1 else 3

expert_trajs = [sample_traj(expert, s) for s in STARTS]

def reached(traj):
    return bool(traj) and traj[-1][2] == GOAL

assert all(reached(traj) for traj in expert_trajs)
assert all(s != GOAL for traj in expert_trajs for s, a, ns in traj)
```

专家优先向下、再向右。选择另一条同长度安全路径并非错误，因此“动作匹配专家”与“任务成功”要分开报告。算法知道环境转移、终止状态和时限；不能声称模型不知道终点在哪里。训练不读取手工奖励，但终止规则本身已提供任务结构。

# 二、Part A：有限时域最大熵 IRL

## 2.1 明确奖励归属与精确动态规划

奖励为 `r_theta(s,a,s_next)=theta[s_next]`：进入一个状态才计一次奖励，进入目标也只计一次。特征为到达状态的 one-hot。对确定性网格中的有限轨迹最大熵模型，后向递推如下：

```text
V_H(s) = 0；V_t(GOAL) = 0
Q_t(s,a) = theta[s_next] + V_{t+1}(s_next)  （仅非终止s）
V_t(s) = logsumexp_a Q_t(s,a)
pi_t(a|s) = exp(Q_t(s,a) - V_t(s))
```

策略有时间索引，因为距离截止还有多少步会影响决策。前向传播非终止状态概率质量，精确得到同一时域下的特征期望；不再用“无限软价值迭代 + 有限采样占用”的不一致组合。

```python
def soft_dp(theta):
    V = np.zeros((H + 1, S))
    pi = np.zeros((H, S, A))
    for t in range(H - 1, -1, -1):
        q = theta[NEXT[:GOAL]] + V[t + 1, NEXT[:GOAL]]
        m = q.max(axis=1, keepdims=True)
        lse = m[:, 0] + np.log(np.exp(q - m).sum(axis=1))
        V[t, :GOAL] = lse
        pi[t, :GOAL] = np.exp(q - lse[:, None])
    return pi, V

def demo_features(trajs):
    mu = np.zeros(S)
    for trajectory in trajs:
        for s, a, ns in trajectory:
            mu[ns] += 1
    return mu / len(trajs)

def expected_features(pi):
    active, mu = p0.copy(), np.zeros(S)
    for t in range(H):
        mass = active[:, None] * pi[t]
        arrivals = np.bincount(NEXT.ravel(), weights=mass.ravel(), minlength=S)
        mu += arrivals  # 与 theta[s_next] 一致；到达目标计一次
        active = arrivals.copy()
        active[GOAL] = 0  # 目标不向外流，也不产生虚构后续动作
    return mu

mu_E = demo_features(expert_trajs)
theta = np.zeros(S)
l2 = 0.01
for iteration in range(200):
    pi, V = soft_dp(theta)
    mu = expected_features(pi)
    # 经验对数似然 - L2；梯度为专家特征减模型特征。
    objective = theta @ mu_E - p0 @ V[0] - 0.5 * l2 * (theta @ theta)
    theta += 0.1 * (mu_E - mu - l2 * theta)
    if iteration % 50 == 0 or iteration == 199:
        print(f"IRL {iteration:3d}: objective={objective:.4f}, feature_error={np.linalg.norm(mu_E-mu):.4f}")

pi, V = soft_dp(theta)
greedy_trajs = [sample_traj(lambda t, s: pi[t, s].argmax(), s) for s in STARTS]
print("IRL 贪心多起点到达率:", np.mean([reached(t) for t in greedy_trajs]))
print("解释示范的奖励矩阵:\n", theta.reshape(N, N))
```

## 2.2 应该怎样解释结果

**预期趋势，不是预填训练日志**：优化目标有机会提高、特征差距趋于减小；有限步长不保证误差每轮单调下降。用任务成功率、路径长度和特征误差分别判断，不要求最高奖励必须在终点，也不能仅凭到达率声称“正确还原真实奖励”。

有限示范、特征选择与奖励整形都会导致多种奖励解释同一行为。L2只是选解偏好，不是消除不可辨识性。此处示范和评测共享起点集合，只验证拟合闭环；更强的迁移结论需要留出起点/改变布局后重新评测，并报告环境变化。

# 三、Part B：表格 GAIL——判别器信号与正确基线

用 `D(s,a)` 表示“来自专家”的概率，训练判别器最大化专家 `log D` 与策略 `log(1-D)`，策略最大化 `log D`。这是一种非饱和代理奖励约定，必须和判别器标签方向配套；不是唯一 GAIL 损失写法。

为了与变长终止轨迹一致，下面统计的是**每回合期望访问次数**，不把专家和策略分别除以各自不同的转移总数。判别器两类贡献按各自回合数平均。未出现的终点动作不参与学习。策略采用表格 logits 与 REINFORCE，而不是原论文的大规模神经网络优化器。

```python
def counts(trajs):
    out = np.zeros((S, A))
    for trajectory in trajs:
        for s, a, ns in trajectory:
            out[s, a] += 1
    return out / len(trajs)

def softmax(x):
    e = np.exp(x - x.max(axis=-1, keepdims=True))
    return e / e.sum(axis=-1, keepdims=True)

expert_counts = counts(expert_trajs)
w = np.zeros((S, A))
pol = np.zeros((H, S, A))
baseline = np.zeros((H, S))  # 只依赖时间/状态，使用之前批次估计
beta, batch_size = 0.01, 48

def evaluate_table(probs, greedy=False, seed=2026):
    g = np.random.default_rng(seed)  # 评测不消耗训练RNG
    trajectories = []
    for start in np.tile(STARTS, 10):
        policy = (lambda t, s: probs[t, s].argmax()) if greedy else (
            lambda t, s: g.choice(A, p=probs[t, s]))
        trajectories.append(sample_traj(policy, start))
    return float(np.mean([reached(t) for t in trajectories]))

for iteration in range(300):
    probs = softmax(pol)
    trajs = [sample_traj(lambda t, s: rng.choice(A, p=probs[t, s]), rng.choice(STARTS))
             for _ in range(batch_size)]
    policy_counts = counts(trajs)
    for _ in range(3):
        D = 1 / (1 + np.exp(-np.clip(w, -30, 30)))
        grad_D = expert_counts * (1 - D) - policy_counts * D
        w += 0.1 * grad_D
    D = 1 / (1 + np.exp(-np.clip(w, -30, 30)))
    rewards = np.log(np.maximum(D, 1e-8))
    entropy = -(probs * np.log(np.maximum(probs, 1e-12))).sum(axis=-1)
    grad, ent_grad = np.zeros_like(pol), np.zeros_like(pol)
    return_sum, visits = np.zeros_like(baseline), np.zeros_like(baseline)
    for trajectory in trajs:
        R = 0.0
        for t in range(len(trajectory) - 1, -1, -1):
            s, a, ns = trajectory[t]
            # 未来熵也进回报；当前熵对logits的显式导数另算。
            R = rewards[s, a] + beta * entropy[t, s] + R
            p = probs[t, s]
            grad_log = -p.copy()
            grad_log[a] += 1
            advantage = R - baseline[t, s]
            grad[t, s] += grad_log * advantage
            ent_grad[t, s] += -p * (np.log(np.maximum(p, 1e-12)) + entropy[t, s])
            return_sum[t, s] += R
            visits[t, s] += 1
    pol += 0.1 * (grad + beta * ent_grad) / batch_size
    mask = visits > 0
    baseline[mask] = 0.9 * baseline[mask] + 0.1 * return_sum[mask] / visits[mask]
    if iteration % 100 == 0 or iteration == 299:
        current = softmax(pol)
        print(f"GAIL {iteration:3d}: stochastic={evaluate_table(current):.3f}, "
              f"greedy={evaluate_table(current, greedy=True):.3f}")
```

基线的关键是逐样本 `grad_log * (R-b(t,s))`。把 `b` 从整个梯度表的每个动作维度直接减去，只会给同一状态的所有 logits 加同一常数，softmax不变，根本没有实现降方差。这里用先前批次的状态—时间基线，避免当前动作直接决定自己的基线；熵梯度单独计算，不与基线混为一项。

**预期观察**：随机策略与贪心策略到达率可能改善，也可能震荡，不能复用旧实现的具体百分比。判别器饱和、示范偏好、有限时域与 `log D` 的负奖励长度偏置都会影响结果；提高熵并不保证成功。需要多随机种子、留出场景和等预算对照才能比较算法优劣。

# 四、必要自检与扩展

```python
# 概率与终止约定
assert np.allclose(pi[:, :GOAL].sum(axis=-1), 1)
assert np.all(pi[:, GOAL] == 0)
assert np.all(V[:, GOAL] == 0)
# 精确占用给出的梯度，应等于有限差分 log-partition 梯度。
eps, index = 1e-5, GOAL
plus, minus = theta.copy(), theta.copy()
plus[index] += eps
minus[index] -= eps
fd = (p0 @ soft_dp(plus)[1][0] - p0 @ soft_dp(minus)[1][0]) / (2 * eps)
assert np.isclose(fd, expected_features(pi)[index], atol=1e-5)
print("终止、概率、特征梯度检查通过")
```

1. 把奖励改成 `theta[s]` 时，要同步修改哪三处：DP、示范特征、模型占用？
2. 用次优示范时，IRL/GAIL为什么也可能忠实复制不良行为？
3. 将部分起点留出，比较动作匹配与到达率；不要把路径不同但同样成功算作失败。
4. 固定采样预算比较有/无基线、有/无熵正则；多次运行才能讨论方差。

# 本实验总结

- IRL示例采用有限时域软DP和时间索引精确占用，终止和奖励归属一致。
- 学到的是解释示范的奖励，不是唯一真实目标；GAIL也使用学习到的替代奖励。
- GAIL基线必须作用在每个样本的回报上，熵项另算，评测区分随机和贪心策略。
- 教学网格结果不代表真实机器人鲁棒性，不能据此断言IRL/GAIL天然优于BC或DAgger。
