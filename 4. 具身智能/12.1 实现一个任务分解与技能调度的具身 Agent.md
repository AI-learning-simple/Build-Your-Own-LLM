# 12.1 实现一个任务分解与技能调度的具身 Agent

> **定位**：Python 标准库 CPU 必修；真实 LLM 接口是选修，真机控制不在本节范围内。
>
> **目标**：实现“计划 → schema/白名单校验 → 前置条件 → 技能 → 后置条件 → 最终目标验收”。空计划、非法技能、超时和缺失目标都必须 **fail-closed（失败关闭）**，不能因为循环没执行就报告成功。

## 一、用可检查状态，而不是打印模拟成功

下面模拟一杯咖啡的流程。它只改变 Python 状态，不控制加热器或机械臂。`pick` 可能失败，其余技能也只有在前置条件成立时才可执行；技能并不一定是学习策略，也可以封装规则控制器。

```python
import json
import random
import time

TASK = "泡一杯咖啡"
SCHEMA = {
    "locate": ("杯子",),
    "pick": ("杯子",),
    "place": ("咖啡机出水口下方",),
    "press_button": ("咖啡机电源",),
    "pour": ("咖啡机", "杯子"),
}


def validate_plan(plan):
    if not isinstance(plan, list) or not 1 <= len(plan) <= 12:
        raise ValueError("计划必须是非空列表，最多 12 步")
    checked = []
    for step in plan:
        if not isinstance(step, dict) or set(step) != {"skill", "args"}:
            raise ValueError("每步只允许 skill 和 args 字段")
        name, args = step["skill"], step["args"]
        if not isinstance(name, str) or name not in SCHEMA:
            raise ValueError("未知技能")
        if (not isinstance(args, list) or not all(isinstance(x, str) for x in args)
                or tuple(args) != SCHEMA[name]):
            raise ValueError("参数数量、类型或枚举值不合法")
        checked.append({"skill": name, "args": args.copy()})
    return checked


def rule_planner(task):
    if task != TASK:
        raise ValueError("不支持的任务，需要人工确认")
    return [{"skill": name, "args": list(args)} for name, args in SCHEMA.items()]


class RobotSkillLibrary:
    def __init__(self, seed=1):
        self.rng = random.Random(seed)
        self.state = {"cup_present": True, "located": False, "holding": None,
                      "cup_location": "桌面", "machine_on": False,
                      "coffee_in_cup": False}
        # 显式绑定；不对任意模型字符串使用 getattr 调度。
        self.skills = {"locate": self.locate, "pick": self.pick, "place": self.place,
                       "press_button": self.press_button, "pour": self.pour}

    def precondition(self, skill):
        s = self.state
        return {
            "locate": s["cup_present"],
            "pick": s["located"] and s["holding"] is None and s["cup_location"] == "桌面",
            "place": s["holding"] == "杯子",
            "press_button": s["cup_location"] == "咖啡机出水口下方" and s["holding"] is None,
            "pour": s["machine_on"] and s["cup_location"] == "咖啡机出水口下方" and not s["coffee_in_cup"],
        }[skill]

    def postcondition(self, skill):
        s = self.state
        return {
            "locate": s["located"],
            "pick": s["holding"] == "杯子",
            "place": s["holding"] is None and s["cup_location"] == "咖啡机出水口下方",
            "press_button": s["machine_on"],
            "pour": s["coffee_in_cup"],
        }[skill]

    def locate(self, obj):
        self.state["located"] = True
        return True

    def pick(self, obj):
        if self.rng.random() < 0.3:
            return False  # 仅本模拟中失败保证无状态变化，因此允许重试
        self.state["holding"] = obj
        self.state["cup_location"] = "夹爪"
        return True

    def place(self, location):
        self.state["cup_location"] = location
        self.state["holding"] = None
        return True

    def press_button(self, button):
        self.state["machine_on"] = True
        return True

    def pour(self, src, dst):
        self.state["coffee_in_cup"] = True
        return True

    def goal_reached(self):
        s = self.state
        return (s["coffee_in_cup"] and s["cup_location"] == "咖啡机出水口下方"
                and s["holding"] is None)
```

这里枚举值非常严格，是教学任务的 schema；扩展新物体需要增加受控参数与状态约束，而不是让模型传入任意 Python 方法或任意坐标。

## 二、有界执行与最终验收

```python
def run_embodied_agent(task, planner=rule_planner, robot=None,
                       max_attempts=3, timeout_s=2.0):
    robot = RobotSkillLibrary() if robot is None else robot
    calls = 0
    deadline = time.monotonic() + timeout_s

    def fail(reason):
        return {"success": False, "reason": reason, "needs_human": True,
                "calls": calls, "state": robot.state.copy()}

    if task != TASK or max_attempts < 1 or timeout_s <= 0:
        return fail("不支持的任务或非法执行预算")
    try:
        plan = validate_plan(planner(task))
    except Exception as exc:
        return fail("规划/解析/校验失败: " + type(exc).__name__)
    if time.monotonic() >= deadline:
        return fail("规划超时")

    for step in plan:
        name, args = step["skill"], step["args"]
        success = False
        # 只有本模拟中无副作用的 pick 失败允许重试。
        attempts = max_attempts if name == "pick" else 1
        for _ in range(attempts):
            if time.monotonic() >= deadline:
                return fail("执行超时")
            if not robot.precondition(name):
                return fail("前置条件失败: " + name)
            try:
                calls += 1
                reported = robot.skills[name](*args)
            except Exception as exc:
                return fail("技能异常: " + type(exc).__name__)
            if time.monotonic() >= deadline:
                return fail("技能返回时已超时")
            if reported and robot.postcondition(name):
                success = True
                break
            if reported:  # 接口声称成功，状态却不符，不盲目重试
                return fail("后置条件失败: " + name)
        if not success:
            return fail("重试预算耗尽: " + name)
    if not robot.goal_reached():
        return fail("步骤已执行，但最终目标未满足")
    return {"success": True, "reason": "最终状态验收通过", "needs_human": False,
            "calls": calls, "state": robot.state.copy()}


# 可运行的最小回归测试，不需要 LLM 或机器人。
assert run_embodied_agent(TASK)["success"]
assert not run_embodied_agent("未知任务")["success"]
assert not run_embodied_agent(TASK, planner=lambda _: [])["success"]
assert not run_embodied_agent(TASK, planner=lambda _: [{"skill": "__init__", "args": []}])["success"]
assert not run_embodied_agent(TASK, planner=lambda _: [{"skill": "pick", "args": ["杯子"]}])["success"]
assert not run_embodied_agent(TASK, planner=lambda _: rule_planner(TASK)[:-1])["success"]
assert not run_embodied_agent(TASK, timeout_s=0)["success"]
print(run_embodied_agent(TASK))
```

**边界**：这里的截止时间检查是合作式的，无法强行打断阻塞的技能或模型调用；生产系统需要独立进程/RPC 超时、底层 watchdog、取消与受控停止，以及真实传感器确认后置条件。这里的字典状态不是视觉/力觉成功检测器，`needs_human=True` 也只是升级请求，不代表已通知到人。

仅在“每次抓取独立成功概率 0.7、失败不改变状态、其他步骤和检测都可靠”的 toy 假设下，最多 3 次尝试成功概率为 `1−0.3³=97.3%`。实际故障可能相关、有副作用，不能把这个数字当真机成功率。本节实现的是局部重试与人工升级，**没有实现自动全局重新规划**。

## 三、接入真实 LLM：提示 JSON 不等于约束解码

提示词要求 JSON 只是软要求。即使推理服务支持 JSON schema/约束解码，也只解决部分结构合法性，仍要做技能权限、参数、可行性及最终目标检查。

下面通过函数注入把接口真正接通：`generate_text(prompt)` 必须由调用者提供，返回文本。它可以封装已有 API 或本地模型；本节不下载模型，不把未执行的调用宣称为测试成功。

```python
def make_llm_planner(generate_text):
    def planner(task):
        prompt = (
            "将任务分解为JSON数组，每项仅有skill和args。不得输出其他文字。\n"
            f"任务: {task}\n允许技能及参数枚举: {json.dumps(SCHEMA, ensure_ascii=False)}"
        )
        response = generate_text(prompt)
        if not isinstance(response, str) or len(response) > 10000:
            raise ValueError("响应类型/长度非法")
        # 不使用 eval；不默默修复非法输出再执行。
        return validate_plan(json.loads(response))
    return planner


# 用已知响应测试适配层与执行器连通；这是 stub 测试，不是真实 LLM 调用。
stub = make_llm_planner(lambda prompt: json.dumps(rule_planner(TASK), ensure_ascii=False))
assert run_embodied_agent(TASK, planner=stub)["success"]
bad_stub = make_llm_planner(lambda prompt: "这不是 JSON")
assert not run_embodied_agent(TASK, planner=bad_stub)["success"]

# result = run_embodied_agent(TASK, planner=real_planner, timeout_s=30)
```

换 Planner 时可复用校验与执行接口，但**不能说无需任何改动**：必须适配模型响应、请求超时、输入观测、技能 schema 与版本，并重新运行非法计划、失败检测和验收测试。模型不支持的任务应拒绝或交人工，不应以空计划“成功”。

## 本实验总结

- 成功来自可检查的最终状态，而不是调用次数或日志。
- schema/白名单、前后置条件、预算与人工路径各有职责，缺一项都可能误报成功。
- 真实机器人还需要时钟、传感器、技能取消和安全控制链；本 CPU 示例不能替代这些工程层。
