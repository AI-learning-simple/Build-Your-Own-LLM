# 8.1 亲手搭建一个 MCP Server，并让 Agent 通过 MCP 协议调用工具

> **本节目标**：第8章讲了为什么 Agent 需要 MCP（Model Context Protocol）这样一个统一的能力连接协议——在MCP 之前，每接入一个新工具，都要给每个 Agent 框架单独写一套适配代码；有了 MCP，工具只需要实现一次协议，就能被任何支持 MCP 的 Agent 调用。这一节我们亲手搭建一个最小的 MCP Server，再写一个客户端通过协议真正调用它。

## 二、搭建一个最小的 MCP Server

创建 `weather_server.py`：

```python
# 用字符串拼接生成一份独立的 Server 脚本文件（MCP Server 通常作为独立进程启动）
server_code = '''
from mcp.server.fastmcp import FastMCP

mcp = FastMCP("weather-server")     # 创建一个 MCP Server 实例，"weather-server" 是它的标识名

FAKE_WEATHER = {
    "北京": "晴，25°C",
    "上海": "多云，28°C",
    "深圳": "小雨，30°C",
}

@mcp.tool()   # 用装饰器把一个普通函数注册成MCP 工具，函数签名和docstring 会自动变成工具的 Schema
def get_weather(city: str) -> str:
    """查询指定城市的天气情况"""
    return FAKE_WEATHER.get(city, f"暂无{city} 的天气数据")

@mcp.tool()
def calculator(expression: str) -> str:
    """计算一个数学表达式"""
    try:
        return str(eval(expression))
    except Exception as e:
        return f"计算出错: {e}"

if __name__ == "__main__":
    mcp.run(transport="stdio")     # 以标准输入输出作为通信方式启动 Server 进程
'''

with open("weather_server.py", "w") as f:
    f.write(server_code)

print("weather_server.py 已生成")
```

`FastMCP` 是 MCP 官方 SDK 提供的高层封装：只需要用 `@mcp.tool()` 装饰一个普通 Python 函数，函数的类型签名和docstring 会自动被转换成 MCP 协议要求的工具描述（Schema），完全不需要手写 JSON Schema。

```bash

```

## 四、亲手写一个 MCP Client 连接这个 Server

```python
client_code = '''
import asyncio
from mcp import ClientSession, StdioServerParameters
from mcp.client.stdio import stdio_client

async def main():
    # 告诉 Client 应该用什么命令启动 Server 进程（这里是运行我们刚生成的 weather_server.py）
    server_params = StdioServerParameters(command="python", args=["weather_server.py"])

    async with stdio_client(server_params) as (read, write):
        async with ClientSession(read, write) as session:
            await session.initialize()          # 完成协议握手

            # 自动发现 Server 提供的全部工具，不需要人工事先知道有哪些工具
            tools = await session.list_tools()
            print("Server 提供的工具:")
            for tool in tools.tools:
                print(f"  - {tool.name}: {tool.description}")

            # 真正调用远程 Server 上的工具，参数以字典形式传入
            result = await session.call_tool("get_weather", {"city": "北京"})
            print("\\n调用 get_weather(城市=北京) 结果:", result.content[0].text)

            result2 = await session.call_tool("calculator", {"expression": "23*17+100"})
            print("调用 calculator(23*17+100) 结果:", result2.content[0].text)

asyncio.run(main())
'''

with open("weather_client.py", "w") as f:
    f.write(client_code)

print("weather_client.py 已生成")
```

```bash
python weather_client.py
```

运行后你会看到 Client 首先通过 `list_tools()` 获取到 Server 暴露的全部工具列表（这一步完全是自动生成的，不需要人工维护"工具清单"），然后分别调用了 `get_weather` 和 `calculator` 两个工具，拿到真实的返回结果——**这就是MCP 协议真正跑通的完整闭环**。

## 五、让 LLM Agent 通过 MCP 调用工具

```python
agent_code = '''
import asyncio
import json
import torch
from transformers import AutoTokenizer, AutoModelForCausalLM
from mcp import ClientSession, StdioServerParameters
from mcp.client.stdio import stdio_client

model_name = "Qwen/Qwen2.5-1.5B-Instruct"
tokenizer = AutoTokenizer.from_pretrained(model_name)
model = AutoModelForCausalLM.from_pretrained(model_name, torch_dtype=torch.float16, device_map="auto")

def llm_decide(question, mcp_tools):
    # 把从 MCP Server 自动发现的工具列表，转换成模型能理解的文字描述
    tool_desc = "\\n".join(f"- {t.name}: {t.description}" for t in mcp_tools)
    prompt = f"""可用工具：
{tool_desc}

用户问题：{question}

请判断需要调用哪个工具，只输出 JSON：{{"tool": "工具名", "arguments": {{...}}}}"""
    messages = [{"role": "user", "content": prompt}]
    text = tokenizer.apply_chat_template(messages, tokenize=False, add_generation_prompt=True)
    inputs = tokenizer(text, return_tensors="pt").to(model.device)
    output = model.generate(**inputs, max_new_tokens=80, do_sample=False)
    return tokenizer.decode(output[0][inputs["input_ids"].shape[1]:], skip_special_tokens=True)

async def main():
    server_params = StdioServerParameters(command="python", args=["weather_server.py"])

    async with stdio_client(server_params) as (read, write):
        async with ClientSession(read, write) as session:
            await session.initialize()
            tools = (await session.list_tools()).tools     # 动态获取工具列表，交给模型做决策

            question = "北京今天天气怎么样？"
            decision_text = llm_decide(question, tools)
            print("模型决策:", decision_text)

            # 把模型的决策解析出来，通过 MCP 协议真正调用对应的工具
            decision = json.loads(decision_text.strip())
            result = await session.call_tool(decision["tool"], decision["arguments"])
            print("MCP 工具调用结果:", result.content[0].text)

asyncio.run(main())
'''

with open("mcp_agent.py", "w") as f:
    f.write(agent_code)

print("mcp_agent.py 已生成，运行: python mcp_agent.py")
```

## 六、MCP 真正解决了什么问题？

对比 2.1 节我们手写的 Function Calling：那里工具的 Schema 是我们手工写在Prompt 里的，工具的执行逻辑和Agent 代码写在同一个文件里。而这一节的 MCP Server 是**完全独立的进程**，任何支持 MCP 协议的客户端（不管是我们自己写的、还是 Claude Desktop、还是其它Agent 框架）都可以通过 `list_tools()` 自动发现它提供了什么能力，不需要为每个 Agent 框架重新适配一遍——**这正是"统一连接协议"的价值：工具开发者只需要实现一次 MCP Server，就能被整个生态复用。**
