## 一、手写版：短期记忆（对话缓冲区）

```python
class ShortTermMemory:
    def __init__(self, max_turns=5):
        self.history = []# 保存 {role, content} 形式的对话记录
        self.max_turns = max_turns      # 最多保留多少轮对话，防止无限增长

    def add(self, role, content):
        self.history.append({"role": role, "content": content})
        # 超过上限后，只保留最近的 max_turns 轮（每轮包含 user + assistant 两条）
        if len(self.history) > self.max_turns * 2:
            self.history = self.history[-self.max_turns * 2:]

    def get_context(self):
        # 把历史对话拼接成一段可以直接放进 Prompt 的文本
        return "\n".join(f"{h['role']}: {h['content']}" for h in self.history)

memory = ShortTermMemory(max_turns=3)
memory.add("user", "我叫小明。")
memory.add("assistant", "你好小明！")
memory.add("user", "我喜欢打篮球。")
print(memory.get_context())
```

短期记忆很简单：**就是一个不断增长（但有上限）的列表**。它能解决"这轮对话记得住上一轮说了什么"的问题，但没办法解决"三天前聊过的事情还记得吗"——这就需要长期记忆。

## 二、手写版：长期记忆（最小向量存储）


```python
import numpy as np
from sentence_transformers import SentenceTransformer

# 用一个多语言的小型句向量模型，把文本转换成语义向量
embedder = SentenceTransformer("paraphrase-multilingual-MiniLM-L12-v2")

class VectorMemory:
    def __init__(self):
        self.texts = []       # 原始文本
        self.vectors = []# 对应的 Embedding 向量

    def remember(self, text):
        vec = embedder.encode(text)
        self.texts.append(text)
        self.vectors.append(vec)

    def recall(self, query, top_k=2):
        if not self.vectors:
            return []
        query_vec = embedder.encode(query)
        vectors = np.array(self.vectors)
        # 计算余弦相似度：两个向量的点积除以各自的模长，值越大代表语义越接近
        sims = vectors @ query_vec / (np.linalg.norm(vectors, axis=1) * np.linalg.norm(query_vec) + 1e-8)
        # 取相似度最高的 top_k 条记忆（argsort 默认升序，用[::-1] 反转成降序）
        top_indices = np.argsort(sims)[::-1][:top_k]
        return [(self.texts[i], float(sims[i])) for i in top_indices]

long_memory = VectorMemory()
long_memory.remember("小明是一名软件工程师，住在北京。")
long_memory.remember("小明最喜欢的运动是篮球。")
long_memory.remember("小红是一名医生，住在上海。")

results = long_memory.recall("小明住在哪里？")
for text, score in results:
    print(f"[相似度 {score:.3f}] {text}")
```

这就是 RAG（检索增强生成）最基础的原理——**用 Embedding 把文字变成向量，用余弦相似度找到最相关的历史记忆，再把它们塞回 Prompt 里**。

## 三、组装成一个带记忆的 Agent

```python
import torch
from transformers import AutoTokenizer, AutoModelForCausalLM

model_name = "Qwen/Qwen2.5-1.5B-Instruct"
tokenizer = AutoTokenizer.from_pretrained(model_name)
model = AutoModelForCausalLM.from_pretrained(model_name, torch_dtype=torch.float16, device_map="auto")

def chat_with_memory(user_input, short_mem, long_mem):
    # 先从长期记忆里检索和当前问题最相关的历史信息
    recalled = long_mem.recall(user_input, top_k=2)
    recalled_text = "\n".join(f"- {t}" for t, _ in recalled)

    # 把"长期记忆检索结果" + "短期对话历史" + "当前问题"一起拼进 Prompt
    prompt = f"""以下是你记得的相关背景信息：
{recalled_text}

最近的对话：
{short_mem.get_context()}

user: {user_input}
assistant:"""

    inputs = tokenizer(prompt, return_tensors="pt").to(model.device)
    output = model.generate(**inputs, max_new_tokens=100, do_sample=False)
    answer = tokenizer.decode(output[0][inputs["input_ids"].shape[1]:], skip_special_tokens=True)

    # 更新短期记忆（这轮对话）和长期记忆（把用户说过的话也存起来，供未来检索）
    short_mem.add("user", user_input)
    short_mem.add("assistant", answer)
    long_mem.remember(f"user说过: {user_input}")

    return answer

short_mem = ShortTermMemory()
answer = chat_with_memory("我住在哪个城市来着？", short_mem, long_memory)
print(answer)
```

## 四、用 LangChain Memory 实现同样的功能

```bash
pip install langchain langchain-community faiss-cpu
```

```python
from langchain.memory import ConversationBufferMemory, VectorStoreRetrieverMemory
from langchain_community.vectorstores import FAISS
from langchain_community.embeddings import HuggingFaceEmbeddings

# ConversationBufferMemory 对应我们手写的 ShortTermMemory，接口更统一
buffer_memory = ConversationBufferMemory()
buffer_memory.save_context({"input": "我叫小明。"}, {"output": "你好小明！"})
buffer_memory.save_context({"input": "我喜欢打篮球。"}, {"output": "篮球是很棒的运动！"})

print(buffer_memory.load_memory_variables({}))
```

```python
# VectorStoreRetrieverMemory 对应我们手写的 VectorMemory，
# 只是把"手算余弦相似度"换成了工业级的 FAISS 向量数据库
embeddings = HuggingFaceEmbeddings(model_name="paraphrase-multilingual-MiniLM-L12-v2")
vectorstore = FAISS.from_texts(
    ["小明是一名软件工程师，住在北京。", "小明最喜欢的运动是篮球。", "小红是一名医生，住在上海。"],
    embeddings,
)
retriever_memory = VectorStoreRetrieverMemory(retriever=vectorstore.as_retriever(search_kwargs={"k": 2}))

print(retriever_memory.load_memory_variables({"prompt": "小明住在哪里？"}))
```

## 五、两种实现方式的对比

| | 手写版 | LangChain Memory |
|---|---|---|
| 短期记忆 | 一个 Python List | `ConversationBufferMemory`（还支持自动摘要压缩的 `ConversationSummaryMemory`） |
| 长期记忆 | 手写余弦相似度计算 | 基于 FAISS/Chroma 等专业向量数据库，检索效率更高 |
| 可插拔性 | 需要自己改代码适配不同 Embedding/存储 | 统一接口，换一个向量数据库只需换一行|
| 学习价值 | 看懂原理 | 直接用于生产环境 |

亲手实现一遍之后再看 LangChain 的 `VectorStoreRetrieverMemory`，你会发现它做的事情和我们手写的 `VectorMemory` 几乎一样：**存文本、算Embedding、找相似、拼回 Prompt**——只是换成了工业级的向量数据库和更完善的接口封装。
