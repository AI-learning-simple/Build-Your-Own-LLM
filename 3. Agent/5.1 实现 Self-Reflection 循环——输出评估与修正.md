# 5.1 实现 Self-Reflection 循环——输出评估与修正

> **本节目标**：第5章讲了Reflection——Agent 如何自我修正。这一节我们亲手实现最经典的 Self-Refine 循环：生成一个答案 → 让模型批评自己的答案 → 根据批评修正答案，观察这个循环是否真的能提升答案质量。

---

## 一、准备模型

```python
import torch
from transformers import AutoTokenizer, AutoModelForCausalLM

model_name = "Qwen/Qwen2.5-1.5B-Instruct"
tokenizer = AutoTokenizer.from_pretrained(model_name)
model = AutoModelForCausalLM.from_pretrained(model_name, torch_dtype=torch.float16, device_map="auto")

def llm(prompt, max_new_tokens=300):
    messages = [{"role": "user", "content": prompt}]
    text = tokenizer.apply_chat_template(messages, tokenize=False, add_generation_prompt=True)
    inputs = tokenizer(text, return_tensors="pt").to(model.device)
    output = model.generate(**inputs, max_new_tokens=max_new_tokens, do_sample=False)
    return tokenizer.decode(output[0][inputs["input_ids"].shape[1]:], skip_special_tokens=True)
```

---

## 二、第一步：生成一个初始答案

```python
question = "小明有一筐鸡蛋，卖掉一半后又卖掉剩下的3个，还剩4个，请问小明原来有多少个鸡蛋？"

# 第一次生成，不做任何检查——这就是普通 Chat 场景下的默认行为
initial_answer = llm(f"请回答下面的问题：\n{question}")
print("=== 初始答案 ===")
print(initial_answer)
```

---

## 三、第二步：让模型批评自己的答案

```python
def critique(question, answer):
    # 关键：这里让模型切换到"评审者"视角，而不是"解题者"视角——
    # 同一个模型，换一个 Prompt 角色，就能对已有答案进行审视
    prompt = f"""请仔细检查下面这个回答是否正确，指出其中的错误（如果有）。如果完全正确，请回答"没有发现错误"。

问题：{question}

回答：{answer}"""
    return llm(prompt, max_new_tokens=200)

feedback = critique(question, initial_answer)
print("\n=== 自我批评 ===")
print(feedback)
```

---

## 四、第三步：根据批评修正答案

```python
def revise(question, answer, feedback):
    # 如果批评环节认为没有问题，直接返回原答案，不需要多此一举
    if "没有发现错误" in feedback or "没有错误" in feedback:
        return answer

    # 否则把"原答案 + 批评意见"一起交给模型，让它重新给出修正后的回答
    prompt = f"""这是之前的回答：
{answer}

这是对这个回答的批评意见：
{feedback}

请根据批评意见，重新给出一个正确的回答：
{question}"""
    return llm(prompt, max_new_tokens=300)

revised_answer = revise(question, initial_answer, feedback)
print("\n=== 修正后的答案 ===")
print(revised_answer)
```

---

## 五、组装成一个完整的 Reflection 循环

```python
def self_refine(question, max_iterations=3):
    answer = llm(f"请回答下面的问题：\n{question}")
    print(f"[初始答案] {answer}\n")

    for i in range(max_iterations):
        feedback = critique(question, answer)
        print(f"[第{i + 1}轮批评] {feedback}\n")

        # 一旦模型认为答案已经没有问题，就提前结束循环，不浪费额外的计算
        if "没有发现错误" in feedback or "没有错误" in feedback:
            print("模型认为答案已经正确，停止迭代。")
            break

        answer = revise(question, answer, feedback)
        print(f"[第{i + 1}轮修正] {answer}\n")

    return answer

final_answer = self_refine(question)
print("========== 最终答案 ==========")
print(final_answer)
```

## 六、验证：Reflection 真的有用吗？

用一个正确答案已知的数学题批量测试，对比"只生成一次"和"经过 Reflection 循环"两种方式的准确率：

```python
import re

test_cases = [
    {"q": "小明有一筐鸡蛋，卖掉一半后又卖掉剩下的3个，还剩4个，请问小明原来有多少个鸡蛋？", "answer": "14"},
    {"q": "一个班级有40名学生，男生比女生多8人，请问男生有多少人？", "answer": "24"},
]

def extract_number(text):
    # 从文本中提取所有数字，取最后一个作为"最终答案"
    numbers = re.findall(r"-?\d+", text)
    return numbers[-1] if numbers else None

direct_correct = 0
refine_correct = 0

for case in test_cases:
    direct = llm(f"请回答下面的问题：\n{case['q']}")            # 对照组：只生成一次，不做 Reflection
    refined = self_refine(case["q"], max_iterations=2)          # 实验组：经过最多2轮 Reflection

    direct_num = extract_number(direct)
    refined_num = extract_number(refined)

    direct_correct += (direct_num == case["answer"])
    refine_correct += (refined_num == case["answer"])

    print(f"题目: {case['q']}")
    print(f"标准答案: {case['answer']}  直接回答: {direct_num}  Reflection后: {refined_num}")
    print("---")

print(f"直接回答准确率: {direct_correct}/{len(test_cases)}")
print(f"Reflection 后准确率: {refine_correct}/{len(test_cases)}")
```

请注意，Reflection 并不是万能的：如果模型本身缺乏识别自己错误的能力（"批评"这一步本身生成了错误的反馈），循环甚至可能让答案变得更差。这也是第5章提到的 Reflection 的真实局限——**它依赖模型本身具备一定的自我评估能力，而这种能力同样来自训练（尤其是 Reasoning Alignment）**。
