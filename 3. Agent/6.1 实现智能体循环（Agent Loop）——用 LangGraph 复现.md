```python
import ast
import json
import math
import operator
import re
import torch
from transformers import AutoTokenizer, AutoModelForCausalLM

model_name = "Qwen/Qwen2.5-1.5B-Instruct"
tokenizer = AutoTokenizer.from_pretrained(model_name)
model = AutoModelForCausalLM.from_pretrained(model_name, torch_dtype=torch.float16, device_map="auto")

def llm(prompt, max_new_tokens=200):
    text = tokenizer.apply_chat_template(
        [{"role": "user", "content": prompt}], tokenize=False, add_generation_prompt=True
    )
    inputs = tokenizer(text, add_special_tokens=False, return_tensors="pt").to(model.device)
    with torch.inference_mode():
        output = model.generate(**inputs, max_new_tokens=max_new_tokens, do_sample=False)
    return tokenizer.decode(output[0][inputs["input_ids"].shape[1]:], skip_special_tokens=True)

_OPS = {
    ast.Add: operator.add, ast.Sub: operator.sub, ast.Mult: operator.mul,
    ast.Div: operator.truediv, ast.FloorDiv: operator.floordiv,
    ast.Mod: operator.mod, ast.Pow: operator.pow,
    ast.USub: operator.neg, ast.UAdd: operator.pos,
}

def safe_eval(expr: str) -> float:
    """有限浮点算术解释器；不是安全沙箱。"""
    if not isinstance(expr, str) or not 0 < len(expr) <= 200:
        raise ValueError("表达式长度必须在 1～200 字符内")
    tree = ast.parse(expr, mode="eval")
    if sum(1 for _ in ast.walk(tree)) > 64:
        raise ValueError("表达式节点过多")

    def bounded(value):
        if type(value) not in (int, float):
            raise ValueError("只允许实数")
        value = float(value)
        if not math.isfinite(value) or abs(value) > 1e12:
            raise ValueError("数值超出有限计算范围")
        return value

    def evaluate(node, depth=0):
        if depth > 16:
            raise ValueError("表达式嵌套过深")
        if isinstance(node, ast.Expression):
            return evaluate(node.body, depth + 1)
        if isinstance(node, ast.Constant):
            return bounded(node.value)
        if isinstance(node, ast.UnaryOp) and type(node.op) in (ast.USub, ast.UAdd):
            return bounded(_OPS[type(node.op)](evaluate(node.operand, depth + 1)))
        if isinstance(node, ast.BinOp) and type(node.op) in _OPS:
            left = evaluate(node.left, depth + 1)
            right = evaluate(node.right, depth + 1)
            if isinstance(node.op, ast.Pow) and (abs(right) > 10 or not right.is_integer()):
                raise ValueError("指数必须是绝对值不超过 10 的整数")
            return bounded(_OPS[type(node.op)](left, right))
        raise ValueError("只允许数字与 + - * / // % ** 运算")

    return evaluate(tree)
```

工具复用 1.1 节的边界：最多 200 字符、64 个 AST 节点、16 层深度、有限中间值与小整数指数；不支持任意精度或任意 Python。处理不可信请求仍需进程级资源限制，不能把它宣传成安全沙箱。

## 二、所有节点都读取真实依赖

`act` 不只接收“一句话步骤”，还接收原始目标和已有观察；否则“把刚才的结果加 100”缺少操作数。记录中区分真实工具返回和模型生成的文字。

```python
from typing import TypedDict

MAX_STEPS = 6  # 预算单位：Plan 决策轮；DONE 也占一轮

class AgentState(TypedDict):
    goal: str
    memory: list[dict]
    next_step: str
    observation: dict
    plan_count: int
    max_steps: int
    status: str

def initial_state(goal, max_steps=MAX_STEPS):
    if type(max_steps) is not int or not 1 <= max_steps <= 20:
        raise ValueError("max_steps 必须为 1～20 的整数")
    return {"goal": goal, "memory": [], "next_step": "", "observation": {},
            "plan_count": 0, "max_steps": max_steps, "status": "running"}

def plan_node(state):
    response = llm(
        f"目标：{state['goal']}\n真实轨迹：{json.dumps(state['memory'], ensure_ascii=False)}\n"
        '给出下一步任务（一句话）。仅当已有结果足以完成目标时，只输出 DONE。', 80
    ).strip()
    done = response == "DONE"
    return {"next_step": response, "plan_count": state["plan_count"] + 1,
            "status": "model_finished" if done else "running"}

def act_node(state):
    response = llm(
        f"原始目标：{state['goal']}\n已有真实轨迹：{json.dumps(state['memory'], ensure_ascii=False)}\n"
        f"当前步骤：{state['next_step']}\n"
        "需要计算时仅输出 calculator[表达式]；否则仅输出 Result: 这一步的文字结果。"
        "不得编造工具 Observation。", 120
    ).strip()
    match = re.fullmatch(r"calculator\[([^\[\]\n]+)\]", response)
    if match:
        try:
            observation = {"kind": "tool", "ok": True, "expression": match.group(1),
                           "result": safe_eval(match.group(1))}
        except Exception as exc:
            observation = {"kind": "tool", "ok": False, "error": str(exc)}
    elif response.startswith("Result:") and response[7:].strip():
        observation = {"kind": "model_text", "ok": True, "result": response[7:].strip()}
    else:
        observation = {"kind": "protocol", "ok": False, "error": "行动格式非法，未执行工具"}
    return {"observation": {"request": response, **observation}}

def reflect_node(state):
    judgement = llm(
        f"目标：{state['goal']}\n之前轨迹：{json.dumps(state['memory'], ensure_ascii=False)}\n"
        f"步骤：{state['next_step']}\n实际观察：{json.dumps(state['observation'], ensure_ascii=False)}\n"
        "仅回答合理或不合理。这只是本步评估，不是最终任务验收。", 20
    ).strip()
    # 严格取值，不能用子串：不合理也包含合理。
    if judgement not in ("合理", "不合理"):
        judgement = "invalid"
    record = {"step": state["next_step"], "observation": state["observation"], "judgement": judgement}
    status = "budget_exhausted" if state["plan_count"] >= state["max_steps"] else "running"
    return {"memory": state["memory"] + [record], "status": status}
```

Observe 由行动节点写入结构化 `observation`，Reflect 是独立节点，完整轨迹在反思后写入 `memory`。最后一轮允许执行 Act/Reflect，再结束，避免“计划数到 6，却只执行了 5 次动作”的偏差。

## 三、普通 Python 版本

```python
class AgentLoop:
    def __init__(self, goal, max_steps=MAX_STEPS):
        self.state = initial_state(goal, max_steps)

    def run(self):
        for _ in range(self.state["max_steps"]):
            self.state.update(plan_node(self.state))
            if self.state["status"] == "model_finished":
                break
            self.state.update(act_node(self.state))
            self.state.update(reflect_node(self.state))
        return self.state

goal = "计算 23 乘以 17，再加上 100，得到最终结果"
hand_result = AgentLoop(goal).run()
print(hand_result["status"], hand_result["memory"])
```

## 四、LangGraph 版本：相同节点、相同预算

```python
from langgraph.graph import StateGraph, START, END

def after_plan(state):
    return "end" if state["status"] == "model_finished" else "act"

def after_reflect(state):
    return "end" if state["status"] == "budget_exhausted" else "plan"

builder = StateGraph(AgentState)
builder.add_node("plan", plan_node)
builder.add_node("act", act_node)
builder.add_node("reflect", reflect_node)
builder.add_edge(START, "plan")
builder.add_conditional_edges("plan", after_plan, {"act": "act", "end": END})
builder.add_edge("act", "reflect")
builder.add_conditional_edges("reflect", after_reflect, {"plan": "plan", "end": END})
app = builder.compile()

graph_result = app.invoke(initial_state(goal), {"recursion_limit": 3 * MAX_STEPS + 5})
print(graph_result["status"], graph_result["memory"])
```

`recursion_limit` 计图的超级步，不是业务动作次数；这里给它留出高于业务预算的余量。两版最多 6 次 Plan、6 次 Act、6 次 Reflect，最多 18 次模型调用；提前 DONE 不会再执行 Act/Reflect。

两版共享转移函数，因此在**相同的模型响应序列**下有相同轨迹与结束原因。先后调用真实模型不保证逐字相同，不能只比较文本就断言图改变了算法。可用固定响应替身分别覆盖立即 DONE、一步后 DONE、始终不 DONE、工具失败四条路径。

## 实验边界与小结

- `model_finished` 是模型自评，不是独立验证；本题可以额外断言工具结果是否为 491，但这不是通用任务判定器。
- `budget_exhausted` 表示预算用尽，不能当作成功；最后一步反思不控制成败，只给下一次规划提供反馈。
- `model_text` 不是工具证据；模型可能在文字结果中幻觉。
- 工具无外部副作用；没有长历史压缩、并发与进程恢复。持久化见 **6.3**，服务化与执行环境见 **第 10、11 章**。
