## 一、手写版：把五个模块拼在一起

```python
import torch
import re
import json
from transformers import AutoTokenizer, AutoModelForCausalLM

model_name = "Qwen/Qwen2.5-1.5B-Instruct"
tokenizer = AutoTokenizer.from_pretrained(model_name)
model = AutoModelForCausalLM.from_pretrained(model_name, torch_dtype=torch.float16, device_map="auto")

def llm(prompt, max_new_tokens=200):
    messages = [{"role": "user", "content": prompt}]
    text = tokenizer.apply_chat_template(messages, tokenize=False, add_generation_prompt=True)
    inputs = tokenizer(text, return_tensors="pt").to(model.device)
    output = model.generate(**inputs, max_new_tokens=max_new_tokens, do_sample=False)
    return tokenizer.decode(output[0][inputs["input_ids"].shape[1]:], skip_special_tokens=True)

import ast
import operator

_OPS = {
    ast.Add: operator.add, ast.Sub: operator.sub, ast.Mult: operator.mul,
    ast.Div: operator.truediv, ast.FloorDiv: operator.floordiv,
    ast.Mod: operator.mod, ast.Pow: operator.pow,
    ast.USub: operator.neg, ast.UAdd: operator.pos,
}

def safe_eval(expr: str) -> float:
    """只允许数字与算术运算，避免 eval 执行任意代码。"""
    if len(expr) > 200:
        raise ValueError("表达式过长")
    def _eval(node):
        if isinstance(node, ast.Expression):
            return _eval(node.body)
        if isinstance(node, ast.Constant) and isinstance(node.value, (int, float)):
            return node.value
        if isinstance(node, ast.BinOp) and type(node.op) in _OPS:
            left, right = _eval(node.left), _eval(node.right)
            if isinstance(node.op, ast.Pow) and abs(right) > 100:
                raise ValueError("指数过大")
            return _OPS[type(node.op)](left, right)
        if isinstance(node, ast.UnaryOp) and type(node.op) in _OPS:
            return _OPS[type(node.op)](_eval(node.operand))
        raise ValueError("只允许数字与 + - * / // % ** 运算")
    return _eval(ast.parse(expr, mode="eval"))

def calculator(expression):
    try:
        return str(safe_eval(expression))
    except Exception as e:
        return f"计算出错: {e}"

tools = {"calculator": calculator}
```

```python
class AgentLoop:
    def __init__(self, goal, max_steps=6):
        self.goal = goal
        self.max_steps = max_steps
        self.memory = []       # 保存每一步的{step, result, judgement}，作为 Plan 阶段的上下文

    def plan(self):
        # Plan：结合目标和已完成的历史，决定"下一步应该做什么"
        prompt = f"""目标：{self.goal}
已完成的记忆：{self.memory}

请给出下一步应该做什么（一句话），如果任务已经完成，请回答"DONE"。"""
        return llm(prompt, max_new_tokens=60)

    def act(self, step_description):
        # Act：判断这一步是否需要调用工具，需要则真正执行，不需要则直接生成结果
        prompt = f"""请针对下面的步骤，判断是否需要使用calculator 工具。
如果需要，只输出：calculator[表达式]
如果不需要，直接给出这一步的结果。

步骤：{step_description}"""
        response = llm(prompt, max_new_tokens=60)
        match = re.search(r"calculator\[(.*?)\]", response)
        if match:
            result = calculator(match.group(1))
            return f"调用了calculator[{match.group(1)}]，结果是 {result}"
        return response

    def reflect(self, step_description, observation):
        # Reflect：检查这一步的执行结果是否合理，为下一轮 Plan 提供参考
        prompt = f"""步骤：{step_description}
执行结果：{observation}

这个结果是否合理地完成了这一步？只回答"合理"或"不合理"。"""
        return llm(prompt, max_new_tokens=20)

    def run(self):
        for i in range(self.max_steps):
            next_step = self.plan()
            print(f"\n[Step {i + 1}] Plan: {next_step}")

            # 用整行匹配而不是子串匹配，避免模型说"还没 DONE"也被误判为完成
            if re.fullmatch(r"\s*DONE\s*[。.!]?\s*", next_step, re.I):
                print("Agent 判断任务已完成。")
                break

            observation = self.act(next_step)
            print(f"[Step {i + 1}] Observation: {observation}")

            judgement = self.reflect(next_step, observation)
            print(f"[Step {i + 1}] Reflect: {judgement}")

            # 把这一轮的完整记录追加进memory，供下一轮Plan 参考
            self.memory.append({"step": next_step, "result": observation, "judgement": judgement})

        return self.memory

agent = AgentLoop(goal="计算 23乘以 17，再加上 100，得到最终结果")
history = agent.run()
```

```python
from typing import TypedDict, List, Dict
from langgraph.graph import StateGraph, END

MAX_STEPS = 8  # 图也必须有自己的步数上限，否则模型可能永远不说 DONE

class AgentState(TypedDict):
    goal: str
    memory: List[Dict]
    next_step: str
    finished: bool
    step_count: int

def plan_node(state: AgentState) -> AgentState:
    # 对应手写版的 plan()，只是从"实例属性"变成了"显式的 state 字典"
    prompt = f"""目标：{state['goal']}
已完成的记忆：{state['memory']}

请给出下一步应该做什么（一句话），如果任务已经完成，请回答"DONE"。"""
    response = llm(prompt, max_new_tokens=60)
    done = bool(re.fullmatch(r"\s*DONE\s*[。.!]?\s*", response, re.I))
    return {
        "next_step": response,
        "finished": done,
        "step_count": state.get("step_count", 0) + 1,
    }

def act_node(state: AgentState) -> AgentState:
    # 对应手写版的 act() + 记录 memory 的逻辑合并在一起
    step = state["next_step"]
    prompt = f"""请针对下面的步骤，判断是否需要使用 calculator 工具。
如果需要，只输出：calculator[表达式]
如果不需要，直接给出这一步的结果。

步骤：{step}"""
    response = llm(prompt, max_new_tokens=60)
    match = re.search(r"calculator\[(.*?)\]", response)
    observation = f"调用了 calculator[{match.group(1)}]，结果是 {calculator(match.group(1))}" if match else response

    new_memory = state["memory"] + [{"step": step, "result": observation}]
    return {"memory": new_memory}

def route(state: AgentState) -> str:
    # 条件路由：模型宣告完成，或达到步数上限，就结束；否则进入 act
    if state.get("finished") or state.get("step_count", 0) >= MAX_STEPS:
        return "end"
    return "act"

graph = StateGraph(AgentState)
graph.add_node("plan", plan_node)
graph.add_node("act", act_node)

graph.set_entry_point("plan")
graph.add_conditional_edges("plan", route, {"act": "act", "end": END})
graph.add_edge("act", "plan")     # act 执行完后固定回到 plan，形成"环"，这正是 Agent Loop 的核心结构

app = graph.compile()
```

```python
result = app.invoke({"goal": "计算 23 乘以 17，再加上 100，得到最终结果", "memory": [], "next_step": "", "finished": False, "step_count": 0})
for item in result["memory"]:
    print(item)
```

## 三、两种实现的核心差异

手写版用 `for` 循环 + `if/break` 控制流程；LangGraph 版用 `add_conditional_edges` + `add_edge("act", "plan")` 显式声明同一个环。两版的功能等价，差别在于这个环是隐藏在 Python 控制流里，还是成为图上一条真实存在的边。

---

## 实验边界

- **"完成"由模型自评**：`plan_node` 只根据模型是否输出 `DONE` 来设置 `finished`，没有独立验证器检查计算结果。模型可能过早宣布完成，也可能在错误结果上宣布完成。
- **步数上限是硬护栏**：`MAX_STEPS` 能防止死循环，但不能保证任务成功；达到上限时应把"未完成"作为结果返回，而不是假装成功。
- **工具只有 calculator**：没有真实副作用工具，因此未涉及幂等、权限与人工确认。
- **记忆只追加不压缩**：长任务下 `memory` 会持续膨胀，生产环境需要摘要或裁剪。
- **单机单进程**：没有并发、持久化与崩溃恢复；这些属于第 7、11 章的内容。

如果要把它做成可评测的版本，至少应增加：独立的结果校验函数、失败原因分类、每步 Trace 记录，以及一组固定测试用例。
