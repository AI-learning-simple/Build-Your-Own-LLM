# 12.1 亲手为 Agent 搭建一个安全执行环境——本地代码沙箱与浏览器自动化

> **本节目标**：第12章讲了 Agent 如何操作数字世界——执行代码、操作浏览器、操作文件系统。这些能力一旦交给一个会"自主决策"的 Agent，风险也随之而来：如果 Agent 生成的代码里有 `os.system("rm -rf /")` 这种命令，直接用 `exec()` 执行是非常危险的。这一节我们亲手搭建两类真实的执行环境：一个有资源限制的本地代码沙箱，以及一个可控的浏览器自动化环境。

---

## 一、为什么不能直接 `exec()`？

```python
dangerous_code = "import os\nos.system('echo 我可以执行任意命令')"

# 千万不要在生产环境这样做：
# exec(dangerous_code)   # exec 会让这段代码拥有和主程序完全相同的权限
```

`exec()` 让生成的代码拥有和主程序完全相同的权限——它能读写任意文件、发起任意网络请求、甚至删除系统文件。Agent 生成的代码即使不是恶意的，也可能因为模型的错误理解（比如误删文件、死循环耗尽资源）造成事故。

---

## 二、方案A：用受限的子进程 + 超时机制

```python
import subprocess

def run_code_sandboxed(code, timeout=5):
    try:
        # 把代码作为独立子进程运行，而不是在当前进程里直接 exec
        result = subprocess.run(
            ["python3", "-c", code],
            capture_output=True,# 捕获子进程的标准输出/错误，而不是直接打印到终端
            text=True,
            timeout=timeout,         # 超过这个时间还没结束，会被强制终止
        )
        return {"stdout": result.stdout, "stderr": result.stderr, "returncode": result.returncode}
    except subprocess.TimeoutExpired:
        # 捕获超时异常，避免死循环代码把整个沙箱函数卡死
        return {"stdout": "", "stderr": "执行超时，已强制终止", "returncode": -1}

safe_code = "print(sum(range(100)))"
print(run_code_sandboxed(safe_code))

infinite_loop_code = "while True:\n    pass"
print(run_code_sandboxed(infinite_loop_code, timeout=3))   # 3秒后会被强制终止，不会无限期卡住
```

子进程 + `timeout` 至少解决了"死循环"这一类问题——超时后 `subprocess.run` 会自动终止子进程，不会让Agent 无限期挂起主程序。

## 三、方案B：限制资源（CPU 时间、内存）

```python
resource_limit_code = '''
import resource
import subprocess
import sys

def limit_resources():
    # 在子进程真正开始执行代码之前调用，限制其最大 CPU 时间（秒）
    resource.setrlimit(resource.RLIMIT_CPU, (5, 5))
    # 限制虚拟内存上限（字节），这里设为200MB
    resource.setrlimit(resource.RLIMIT_AS, (200 * 1024 * 1024, 200 * 1024 * 1024))

def run_with_limits(code):
    return subprocess.run(
        [sys.executable, "-c", code],
        preexec_fn=limit_resources,   # 在子进程 fork 后、执行目标代码前调用，设置资源限制
        capture_output=True,
        text=True,
        timeout=10,
    )

if __name__ == "__main__":
    result = run_with_limits("print('hello from sandbox')")
    print(result.stdout)
'''

with open("resource_sandbox.py", "w") as f:
    f.write(resource_limit_code)

print("resource_sandbox.py 已生成（注意：resource 模块仅在 Linux/macOS 上可用）")
```

`resource.setrlimit` 可以给子进程设置 CPU 时间上限（5秒）和内存上限（200MB）——即使 Agent 生成的代码存在Bug 导致资源消耗过大，也会被操作系统层面直接终止，而不会影响到主进程。

## 四、把沙箱接入 Agent

```python
import torch
from transformers import AutoTokenizer, AutoModelForCausalLM

model_name = "Qwen/Qwen2.5-1.5B-Instruct"
tokenizer = AutoTokenizer.from_pretrained(model_name)
model = AutoModelForCausalLM.from_pretrained(model_name, torch_dtype=torch.float16, device_map="auto")

def llm(prompt, max_new_tokens=150):
    messages = [{"role": "user", "content": prompt}]
    text = tokenizer.apply_chat_template(messages, tokenize=False, add_generation_prompt=True)
    inputs = tokenizer(text, return_tensors="pt").to(model.device)
    output = model.generate(**inputs, max_new_tokens=max_new_tokens, do_sample=False)
    return tokenizer.decode(output[0][inputs["input_ids"].shape[1]:], skip_special_tokens=True)

def code_agent(task):
    prompt = f"请写一段 Python 代码完成下面的任务，只输出代码，不要解释：\n{task}"
    code = llm(prompt)
    # 模型有时会习惯性地把代码包在Markdown 代码块里，这里去掉多余的围栏标记
    fence = chr(96) * 3    # 用 chr(96) 拼出反引号，避免这段说明文字里出现真正的三重反引号
    code = code.replace(fence + "python", "").replace(fence, "").strip()

    print("生成的代码:")
    print(code)

    # 关键：交给沙箱执行，而不是直接 exec，即使代码有问题也不会影响主程序
    result = run_code_sandboxed(code, timeout=5)
    print("\n沙箱执行结果:")
    print(result["stdout"] or result["stderr"])
    return result

code_agent("计算1到100的和")
```

即使模型生成的代码存在问题，也永远在一个隔离的子进程里运行，主程序始终是安全的。

## 五、浏览器自动化：另一种"操作数字世界"的执行环境

```bash
pip install playwright
playwright install chromium
```

```python
from playwright.sync_api import sync_playwright

def browser_agent_demo(url, action_description):
    with sync_playwright() as p:
        # headless=True 表示不显示真实浏览器窗口，适合在服务器上无界面运行
        browser = p.chromium.launch(headless=True)
        page = browser.new_page()
        page.goto(url)

        title = page.title()
        print(f"页面标题: {title}")

        # locator 用 CSS 选择器定位页面元素，这里抓取所有一级、二级标题的文本
        headings = page.locator("h1, h2").all_inner_texts()
        print(f"页面主要标题: {headings[:5]}")

        browser.close()   # 用完及时关闭浏览器进程，释放资源
        return {"title": title, "headings": headings[:5]}

browser_agent_demo("https://example.com", "获取页面标题和主要标题")
```

这里的"安全"体现在：浏览器运行在一个**独立的、受控的进程**里（`headless=True` 的无头模式），Agent 只能通过 Playwright 提供的 API（`goto`、`click`、`fill` 等）进行操作，而不是直接拥有操作系统级别的权限——这和执行任意 Shell 命令是完全不同的风险等级。

## 六、把浏览器操作接入 Agent

```python
def browser_action_agent(task, url):
    with sync_playwright() as p:
        browser = p.chromium.launch(headless=True)
        page = browser.new_page()
        page.goto(url)

        # 让模型基于当前页面信息，决定接下来要执行哪一种预先定义好的安全操作
        prompt = f"""页面标题是：{page.title()}
任务：{task}

请判断应该执行以下哪个操作：GET_TITLE（获取标题）、GET_HEADINGS（获取标题列表）。只回答操作名称。"""
        action = llm(prompt, max_new_tokens=20).strip()

        # 注意：这里只允许执行两种预先定义的安全操作，模型不能"自由发挥"执行任意浏览器指令
        if "HEADINGS" in action:
            result = page.locator("h1, h2, h3").all_inner_texts()
        else:
            result = page.title()

        browser.close()
        return result

result = browser_action_agent("获取这个页面所有的标题", "https://example.com")
print(result)
```
