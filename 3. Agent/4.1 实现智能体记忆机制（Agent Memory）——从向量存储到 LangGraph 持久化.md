## 一、手写版：短期记忆（对话缓冲区）

```python
class ShortTermMemory:
    def __init__(self, max_turns=5):
        self.history = []# 保存 {role, content} 形式的对话记录
        self.max_turns = max_turns      # 最多保留多少轮对话，防止无限增长

    def add(self, role, content):
        self.history.append({"role": role, "content": content})
        # 超过上限后，只保留最近的 max_turns 轮（每轮包含 user + assistant 两条）
        if len(self.history) > self.max_turns * 2:
            self.history = self.history[-self.max_turns * 2:]

    def get_context(self):
        # 把历史对话拼接成一段可以直接放进 Prompt 的文本
        return "\n".join(f"{h['role']}: {h['content']}" for h in self.history)

memory = ShortTermMemory(max_turns=3)
memory.add("user", "我叫小明。")
memory.add("assistant", "你好小明！")
memory.add("user", "我喜欢打篮球。")
print(memory.get_context())
```

短期记忆很简单：**就是一个不断增长（但有上限）的列表**。它能解决"这轮对话记得住上一轮说了什么"的问题，但没办法解决"三天前聊过的事情还记得吗"——这就需要长期记忆。

## 二、手写版：长期记忆（最小向量存储）

```python
import numpy as np
from sentence_transformers import SentenceTransformer

# 用一个多语言的小型句向量模型，把文本转换成语义向量
embedder = SentenceTransformer("paraphrase-multilingual-MiniLM-L12-v2")

class VectorMemory:
    def __init__(self):
        self.texts = []       # 原始文本
        self.vectors = []# 对应的 Embedding 向量

    def remember(self, text):
        vec = embedder.encode(text)
        self.texts.append(text)
        self.vectors.append(vec)

    def recall(self, query, top_k=2):
        if not self.vectors:
            return []
        query_vec = embedder.encode(query)
        vectors = np.array(self.vectors)
        # 计算余弦相似度：两个向量的点积除以各自的模长，值越大代表语义越接近
        sims = vectors @ query_vec / (np.linalg.norm(vectors, axis=1) * np.linalg.norm(query_vec) + 1e-8)
        # 取相似度最高的 top_k 条记忆（argsort 默认升序，用[::-1] 反转成降序）
        top_indices = np.argsort(sims)[::-1][:top_k]
        return [(self.texts[i], float(sims[i])) for i in top_indices]

long_memory = VectorMemory()
long_memory.remember("小明是一名软件工程师，住在北京。")
long_memory.remember("小明最喜欢的运动是篮球。")
long_memory.remember("小红是一名医生，住在上海。")

results = long_memory.recall("小明住在哪里？")
for text, score in results:
    print(f"[相似度 {score:.3f}] {text}")
```

这段 `recall` 就是长期记忆的读取路径：把查询和记忆都编码成向量，用余弦相似度取 top-k，再把结果拼回 Prompt（RAG 的完整实现见 4.2）。

## 三、带记忆 Agent 的组装

```python
import torch
from transformers import AutoTokenizer, AutoModelForCausalLM

model_name = "Qwen/Qwen2.5-1.5B-Instruct"
tokenizer = AutoTokenizer.from_pretrained(model_name)
model = AutoModelForCausalLM.from_pretrained(model_name, torch_dtype=torch.float16, device_map="auto")

def chat_with_memory(user_input, short_mem, long_mem):
    # 先从长期记忆里检索和当前问题最相关的历史信息
    recalled = long_mem.recall(user_input, top_k=2)
    recalled_text = "\n".join(f"- {t}" for t, _ in recalled)

    # 把"长期记忆检索结果" + "短期对话历史" + "当前问题"一起拼进 Prompt
    prompt = f"""以下是你记得的相关背景信息：
{recalled_text}

最近的对话：
{short_mem.get_context()}

user: {user_input}
assistant:"""

    inputs = tokenizer(prompt, return_tensors="pt").to(model.device)
    output = model.generate(**inputs, max_new_tokens=100, do_sample=False)
    answer = tokenizer.decode(output[0][inputs["input_ids"].shape[1]:], skip_special_tokens=True)

    # 更新短期记忆（这轮对话）和长期记忆（把用户说过的话也存起来，供未来检索）
    short_mem.add("user", user_input)
    short_mem.add("assistant", answer)
    long_mem.remember(f"user说过: {user_input}")

    return answer

short_mem = ShortTermMemory()
answer = chat_with_memory("我住在哪个城市来着？", short_mem, long_memory)
print(answer)
```

## 四、用 LangGraph 的持久化机制实现同样的功能

先说明为什么不再用 LangChain 的 Memory 模块：`langchain.memory` 里的 `ConversationBufferMemory`、`VectorStoreRetrieverMemory` 这一批类，在 LangChain 1.0（2025 年 10 月）的重构中已被移除。官方给出的迁移路径就是 LangGraph 的两组持久化原语：

| 记忆类型 | 对应原语 | 作用范围 |
|---|---|---|
| 短期记忆 | **Checkpointer**（`InMemorySaver` / `SqliteSaver` / `PostgresSaver`） | 按 `thread_id` 保存整张图的 State，同一会话可续跑、可回放 |
| 长期记忆 | **Store**（`InMemoryStore` / `PostgresStore`） | 跨 `thread_id`，按 `namespace` 存事实与偏好，可选向量索引做语义检索 |


```python
from typing import Annotated, TypedDict
from langchain_core.messages import AnyMessage, HumanMessage, AIMessage
from langgraph.graph import StateGraph, START, END
from langgraph.graph.message import add_messages
from langgraph.checkpoint.memory import InMemorySaver

class ChatState(TypedDict):
    # add_messages 是一个 reducer：新消息追加进列表，而不是覆盖整个字段
    messages: Annotated[list[AnyMessage], add_messages]

def generate(messages):
    """把 LangChain 的消息对象转成 chat template 需要的格式，再调用本地模型。"""
    chat = [{"role": "user" if m.type == "human" else "assistant", "content": m.content} for m in messages]
    text = tokenizer.apply_chat_template(chat, tokenize=False, add_generation_prompt=True)
    inputs = tokenizer(text, return_tensors="pt").to(model.device)
    out = model.generate(**inputs, max_new_tokens=100, do_sample=False)
    return tokenizer.decode(out[0][inputs["input_ids"].shape[1]:], skip_special_tokens=True)

def chat_node(state: ChatState):
    return {"messages": [AIMessage(content=generate(state["messages"]))]}

builder = StateGraph(ChatState)
builder.add_node("chat", chat_node)
builder.add_edge(START, "chat")
builder.add_edge("chat", END)

# 短期记忆 = Checkpointer：每执行完一步，整个 State 都会按 thread_id 存一份
graph = builder.compile(checkpointer=InMemorySaver())

config = {"configurable": {"thread_id": "user-1"}}   # 同一个 thread_id 才共享同一段记忆

graph.invoke({"messages": [HumanMessage(content="我叫小明，我喜欢打篮球。")]}, config)
result = graph.invoke({"messages": [HumanMessage(content="我叫什么名字？")]}, config)
print(result["messages"][-1].content)
```

这里没有一行"手动把历史拼进 Prompt"的代码：`HumanMessage` 进入图时，Checkpointer 先把该 `thread_id` 的历史状态取回来，再用 `add_messages` 与新消息合并，节点拿到的 `state["messages"]` 已经是完整的对话。

```python
from langgraph.store.memory import InMemoryStore

# 长期记忆 = Store：按 namespace 分层（这里用"用户 + 事实"），与 thread 无关
store = InMemoryStore()

store.put(("xiaoming", "facts"), "home", {"text": "小明是一名软件工程师，住在北京。"})
store.put(("xiaoming", "facts"), "sport", {"text": "小明最喜欢的运动是篮球。"})
store.put(("xiaohong", "facts"), "job", {"text": "小红是一名医生，住在上海。"})

for item in store.search(("xiaoming", "facts")):    # 按命名空间列举
    print(item.key, item.value["text"])
```

```python
# 配置向量索引后，search 才支持 query 语义检索；不配置索引时只能按命名空间/键过滤
store_with_index = InMemoryStore(
    index={"embed": lambda texts: embedder.encode(texts).tolist(), "dims": 384, "fields": ["text"]}
)
store_with_index.put(("xiaoming", "facts"), "home", {"text": "小明是一名软件工程师，住在北京。"})
store_with_index.put(("xiaoming", "facts"), "sport", {"text": "小明最喜欢的运动是篮球。"})

for item in store_with_index.search(("xiaoming", "facts"), query="小明住在哪里？", limit=1):
    print(item.value["text"])
```

`namespace` 是隔离边界：`("xiaoming", "facts")` 与 `("xiaohong", "facts")` 互不可见——这正好对应第 4 章强调的"记忆必须按用户/租户隔离"。

## 五、两种实现方式的对比

| | 手写版 | LangGraph 持久化 |
|---|---|---|
| 短期记忆 | 一个 Python List，手动裁剪 | Checkpointer 按 `thread_id` 保存整个 State（可换 SQLite/Postgres） |
| 长期记忆 | 手写余弦相似度计算 | Store 按 `namespace` 存取，可配置 Embedding 做语义检索 |
| 可插拔性 | 需要自己改代码适配不同 Embedding/存储 | 换存储后端只需换 Saver/Store 实现 |
| 附带能力 | 无 | 断点恢复、时间旅行回放、人工介入（第 7 章）都与 Checkpointer 共用同一份状态 |
| 学习价值 | 看懂原理 | 直接用于生产环境 |

亲手实现之后再对照这两组原语会发现：**框架并没有替你决定"记什么"**。Checkpointer 只是忠实地把每一步 State 存下来（哪怕里面全是噪声），Store 只是提供带命名空间和可选向量索引的键值存取——写入策略、摘要压缩、冲突更新、遗忘曲线，仍然要自己设计。这也解释了为什么"记忆"是工程问题，而不是换个 API 就能解决的问题。

## 实验边界

- 向量存储与对话历史都在内存中，进程退出即丢失（换成 `SqliteSaver`/`PostgresStore` 才能持久化）；
- 使用简化向量化/相似度实现，不代表生产级 Embedding 与检索质量；
- 没有设计写入策略、摘要、冲突更新与遗忘；
- 只演示“取回相关历史”，没有评估记忆污染与错误传播；
- Checkpointer 与 Store 是状态管理原语，不含记忆策略；框架版本仍可能调整参数名与默认行为。

## 本节小结

- 手写实现了短期记忆（对话缓冲）和长期记忆（向量检索）两种最基础的记忆机制
- 组装出一个能"记住之前聊过的事情"的最小 Agent
- 用 LangGraph 的 Checkpointer 与 Store 复现了同样的功能，理解了框架封装的价值。一句话总结：**Agent 的记忆本质上就是"存储 + 检索"，短期记忆解决"这轮对话"，长期记忆（向量检索）解决"跨对话、跨会话"的信息留存。**
