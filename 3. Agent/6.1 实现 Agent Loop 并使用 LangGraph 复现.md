## 一、手写版：把五个模块拼在一起

```python
import torch
import re
import json
from transformers import AutoTokenizer, AutoModelForCausalLM

model_name = "Qwen/Qwen2.5-1.5B-Instruct"
tokenizer = AutoTokenizer.from_pretrained(model_name)
model = AutoModelForCausalLM.from_pretrained(model_name, torch_dtype=torch.float16, device_map="auto")

def llm(prompt, max_new_tokens=200):
    messages = [{"role": "user", "content": prompt}]
    text = tokenizer.apply_chat_template(messages, tokenize=False, add_generation_prompt=True)
    inputs = tokenizer(text, return_tensors="pt").to(model.device)
    output = model.generate(**inputs, max_new_tokens=max_new_tokens, do_sample=False)
    return tokenizer.decode(output[0][inputs["input_ids"].shape[1]:], skip_special_tokens=True)

def calculator(expression):
    try:
        return str(eval(expression))
    except Exception as e:
        return f"计算出错: {e}"

tools = {"calculator": calculator}
```

```python
class AgentLoop:
    def __init__(self, goal, max_steps=6):
        self.goal = goal
        self.max_steps = max_steps
        self.memory = []       # 保存每一步的{step, result, judgement}，作为 Plan 阶段的上下文

    def plan(self):
        # Plan：结合目标和已完成的历史，决定"下一步应该做什么"
        prompt = f"""目标：{self.goal}
已完成的记忆：{self.memory}

请给出下一步应该做什么（一句话），如果任务已经完成，请回答"DONE"。"""
        return llm(prompt, max_new_tokens=60)

    def act(self, step_description):
        # Act：判断这一步是否需要调用工具，需要则真正执行，不需要则直接生成结果
        prompt = f"""请针对下面的步骤，判断是否需要使用calculator 工具。
如果需要，只输出：calculator[表达式]
如果不需要，直接给出这一步的结果。

步骤：{step_description}"""
        response = llm(prompt, max_new_tokens=60)
        match = re.search(r"calculator\[(.*?)\]", response)
        if match:
            result = calculator(match.group(1))
            return f"调用了calculator[{match.group(1)}]，结果是 {result}"
        return response

    def reflect(self, step_description, observation):
        # Reflect：检查这一步的执行结果是否合理，为下一轮 Plan 提供参考
        prompt = f"""步骤：{step_description}
执行结果：{observation}

这个结果是否合理地完成了这一步？只回答"合理"或"不合理"。"""
        return llm(prompt, max_new_tokens=20)

    def run(self):
        for i in range(self.max_steps):
            next_step = self.plan()
            print(f"\n[Step {i + 1}] Plan: {next_step}")

            # 一旦 Plan 判断任务已完成，立即退出循环
            if "DONE" in next_step:
                print("Agent 判断任务已完成。")
                break

            observation = self.act(next_step)
            print(f"[Step {i + 1}] Observation: {observation}")

            judgement = self.reflect(next_step, observation)
            print(f"[Step {i + 1}] Reflect: {judgement}")

            # 把这一轮的完整记录追加进memory，供下一轮Plan 参考
            self.memory.append({"step": next_step, "result": observation, "judgement": judgement})

        return self.memory

agent = AgentLoop(goal="计算 23乘以 17，再加上 100，得到最终结果")
history = agent.run()
```

```python
from typing import TypedDict, List, Dict
from langgraph.graph import StateGraph, END

class AgentState(TypedDict):
    goal: str
    memory: List[Dict]
    next_step: str
    finished: bool

def plan_node(state: AgentState) -> AgentState:
    # 对应手写版的 plan()，只是从"实例属性"变成了"显式的 state 字典"
    prompt = f"""目标：{state['goal']}
已完成的记忆：{state['memory']}

请给出下一步应该做什么（一句话），如果任务已经完成，请回答"DONE"。"""
    response = llm(prompt, max_new_tokens=60)
    return {"next_step": response, "finished": "DONE" in response}

def act_node(state: AgentState) -> AgentState:
    # 对应手写版的 act() + 记录 memory 的逻辑合并在一起
    step = state["next_step"]
    prompt = f"""请针对下面的步骤，判断是否需要使用 calculator 工具。
如果需要，只输出：calculator[表达式]
如果不需要，直接给出这一步的结果。

步骤：{step}"""
    response = llm(prompt, max_new_tokens=60)
    match = re.search(r"calculator\[(.*?)\]", response)
    observation = f"调用了 calculator[{match.group(1)}]，结果是 {calculator(match.group(1))}" if match else response

    new_memory = state["memory"] + [{"step": step, "result": observation}]
    return {"memory": new_memory}

def route(state: AgentState) -> str:
    # 条件路由：finished 为 True 就结束，否则回到 act 节点继续执行
    return "end" if state["finished"] else "act"

graph = StateGraph(AgentState)
graph.add_node("plan", plan_node)
graph.add_node("act", act_node)

graph.set_entry_point("plan")
graph.add_conditional_edges("plan", route, {"act": "act", "end": END})
graph.add_edge("act", "plan")     # act 执行完后固定回到 plan，形成"环"，这正是 Agent Loop 的核心结构

app = graph.compile()
```

```python
result = app.invoke({"goal": "计算 23 乘以 17，再加上 100，得到最终结果", "memory": [], "next_step": "", "finished": False})
for item in result["memory"]:
    print(item)
```

## 三、两种实现的核心差异

手写版的 `AgentLoop.run()` 用一个普通的 `for` 循环 + `if/break` 来控制流程；LangGraph 版用 `add_conditional_edges` + `add_edge("act", "plan")` 显式声明了一个"环"（`plan → act → plan → act → ... → end`）。这个"环"正是 Agent Loop 的本质：**它不是一条直线，而是一个会不断回到起点重新评估的循环，直到某个条件触发退出**。手写版把这个环隐藏在 Python 的控制流里，LangGraph 版把这个环变成了图结构上一条真实存在的边——这也是为什么 LangGraph 特别适合搭建 Agent：**Agent 的核心特征就是"循环"，而 Graph 天生就是用来表达循环和分支的数据结构。**
