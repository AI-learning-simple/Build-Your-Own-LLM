# 2.1 实现函数调用（Function Calling）——从 JSON 解析到原生工具调用（Tool Calling）

> **本节定位**：原理见第 2 章。这里比较手写 JSON 协议与 Qwen2.5 本地原生工具模板，两种方式都完成“生成请求 → 校验 → 执行一次 → 回填 → 回答”。
>
> **硬件要求**：本地推理建议 8GB+ 显存；小模型可能不遵循格式。
>
> **边界**：这是一个**最小单调用**实验，不是兼容所有模型/OpenAI 服务的通用协议。多调用、流式分片与自动纠错循环不在本节范围内。

## 一、共同准备：同一模型、同一 chat template

按顺序运行代码块。A/B 两路使用同一问题、模型、生成参数和 `apply_chat_template`；区别仅在工具协议的表达，不把 A 的裸文本和 B 的聊天模板混比。

```python
import json
import re
import torch
from transformers import AutoTokenizer, AutoModelForCausalLM

model_name = "Qwen/Qwen2.5-1.5B-Instruct"
tokenizer = AutoTokenizer.from_pretrained(model_name)
model = AutoModelForCausalLM.from_pretrained(
    model_name, torch_dtype=torch.float16, device_map="auto"
)

def generate(messages, tool_schema=None):
    kwargs = {"tools": tool_schema} if tool_schema is not None else {}
    text = tokenizer.apply_chat_template(
        messages, tokenize=False, add_generation_prompt=True, **kwargs
    )
    inputs = tokenizer(text, add_special_tokens=False, return_tensors="pt").to(model.device)
    with torch.inference_mode():
        output = model.generate(**inputs, max_new_tokens=160, do_sample=False)
    return tokenizer.decode(output[0][inputs["input_ids"].shape[1]:], skip_special_tokens=True)

def get_weather(city):
    # 固定模拟数据，不是今天的真实天气。
    return {"北京": "晴，25°C", "上海": "多云，28°C"}.get(city, "未知城市")

def failure(code, message):
    return {"ok": False, "error": {"code": code, "message": message}}

def execute_once(call):
    """服务端白名单与参数校验；任何校验失败都不能调用函数。"""
    if not isinstance(call, dict) or set(call) != {"name", "arguments"}:
        return failure("invalid_call", "调用必须且只能包含 name 与 arguments")
    if not isinstance(call["name"], str) or call["name"] != "get_weather":
        return failure("unknown_tool", "仅允许 get_weather")
    args = call["arguments"]
    if not isinstance(args, dict) or set(args) != {"city"}:
        return failure("invalid_arguments", "必须且只能提供 city 参数")
    if not isinstance(args["city"], str) or not args["city"].strip() or len(args["city"]) > 100:
        return failure("invalid_arguments", "city 必须为 1～100 字符的非空字符串")
    try:
        result = get_weather(city=args["city"].strip())  # 唯一真实执行点
        return {"ok": True, "result": result}
    except Exception as exc:
        return failure("tool_error", str(exc))

question = "北京的模拟天气怎么样？"
```

`required` 只声明必填，不会自动在 Python 执行端验证类型；上面同时拒绝未知工具、非字符串 city、额外参数和空参数。

## 二、方式 A：手写 JSON 协议

此处要求模型要么生成一个调用，要么生成一个 `answer` 对象。没有使用 OpenAI `tool` 消息协议，结果作为明确标注的程序观察数据回填。

```python
manual_instruction = """可用函数 get_weather(city: str)，查询固定模拟天气。
需要工具时只输出 JSON：{"function":"get_weather","arguments":{"city":"北京"}}
不需要工具时只输出 JSON：{"answer":"直接回答"}
不要 Markdown，不要额外字段。"""

def parse_manual_output(text):
    try:
        obj = json.loads(text.strip())
    except (json.JSONDecodeError, TypeError):
        return "error", failure("parse_error", "不是合法 JSON")
    if isinstance(obj, dict) and set(obj) == {"answer"} and isinstance(obj["answer"], str) and obj["answer"].strip():
        return "answer", obj["answer"]
    if isinstance(obj, dict) and set(obj) == {"function", "arguments"}:
        return "call", {"name": obj["function"], "arguments": obj["arguments"]}
    return "error", failure("invalid_call", "不是单调用或 answer 对象")

def run_manual(question):
    manual_messages = [
        {"role": "system", "content": manual_instruction},
        {"role": "user", "content": question},
    ]
    manual_raw = generate(manual_messages)
    kind, payload = parse_manual_output(manual_raw)
    if kind == "answer":
        return {"status": "direct_answer", "answer": payload}
    manual_result = execute_once(payload) if kind == "call" else payload
    manual_messages.extend([
        {"role": "assistant", "content": manual_raw},
        {"role": "user", "content": "程序观察（JSON 数据）：" + json.dumps(manual_result, ensure_ascii=False)
         + '\n工具阶段已结束。仅输出 {"answer":"答案或失败说明"}，不能再请求工具。'},
    ])
    final_kind, answer = parse_manual_output(generate(manual_messages))
    return {"status": "answered" if final_kind == "answer" else "unfinished",
            "observation": manual_result,
            "answer": answer if final_kind == "answer" else "收尾格式不合法；未执行新调用"}

manual_run = run_manual(question)
print("方式 A：", manual_run)
```

解析失败也产生结构化 Observation，但本节不自动重试；即使收尾再次提出调用，也不执行。

## 三、方式 B：Qwen2.5 原生模板

工具 Schema 使用常见 function 定义格式；是否支持该格式取决于模型和模板，不能仅靠传 `tools` 就让任意模型获得工具能力。

```python
tool_schema = [{
    "type": "function",
    "function": {
        "name": "get_weather",
        "description": "查询指定城市的固定模拟天气，不是实时天气",
        "parameters": {
            "type": "object",
            "properties": {"city": {"type": "string", "minLength": 1, "maxLength": 100}},
            "required": ["city"],
            "additionalProperties": False,
        },
    },
}]

def parse_native_tool_call(text):
    """只接受一个完整 Qwen tool_call 块或一个裸 JSON 对象。"""
    text = text.strip()
    if not text:
        return "error", failure("empty_output", "模型没有输出")
    if "<tool_call>" in text or "</tool_call>" in text:
        if text.count("<tool_call>") != 1 or text.count("</tool_call>") != 1:
            return "error", failure("unsupported_calls", "只支持一个调用，多个或破损标签均拒绝")
        match = re.fullmatch(r"<tool_call>\s*(.*?)\s*</tool_call>", text, re.S)
        if not match:
            return "error", failure("parse_error", "调用标签外存在额外内容")
        raw = match.group(1)
    elif text.startswith(("{", "[")):
        raw = text
    else:
        return "answer", text   # 普通回答不是空工具调用，不应追加孤立 tool 消息
    try:
        obj = json.loads(raw)
    except json.JSONDecodeError:
        return "error", failure("parse_error", "调用体不是合法 JSON")
    if not isinstance(obj, dict) or set(obj) != {"name", "arguments"} or not isinstance(obj["name"], str):
        return "error", failure("invalid_call", "仅接受 name/arguments 单调用对象")
    return "call", obj

def run_native(question):
    native_messages = [{"role": "user", "content": question}]
    native_raw = generate(native_messages, tool_schema)
    kind, payload = parse_native_tool_call(native_raw)
    if kind == "answer":
        return {"status": "direct_answer", "answer": payload}
    if kind == "call":
        native_result = execute_once(payload)  # 保存结果，回填时绝不重新执行
        native_messages.extend([
            {"role": "assistant", "content": "", "tool_calls": [
                {"type": "function", "function": payload}
            ]},
            {"role": "tool", "name": payload["name"],
             "content": json.dumps(native_result, ensure_ascii=False)},
        ])
    else:
        native_result = payload
        # 解析不出合法请求时没有可配对的调用；以程序错误数据回填，而非伪造 tool_call。
        native_messages.extend([
            {"role": "assistant", "content": native_raw},
            {"role": "user", "content": "程序解析结果：" + json.dumps(native_result, ensure_ascii=False)},
        ])
    native_messages.append({"role": "user", "content":
        "工具阶段已结束。根据真实观察回答；若失败就说明失败。禁止新增调用。"})
    final_kind, answer = parse_native_tool_call(generate(native_messages, tool_schema))
    return {"status": "answered" if final_kind == "answer" else "unfinished",
            "observation": native_result,
            "answer": answer if final_kind == "answer" else "模型仍请求工具或输出非法；本例停止，不执行"}

native_run = run_native(question)
print("方式 B：", native_run)
```

### 本地模板与 OpenAI 服务协议不是一回事

Qwen2.5 本地模板把 `tool_calls[].function.arguments` 的 **dict** 序列化到 `<tool_call>`，把 `role="tool"` 消息包进 `<tool_response>`；本例不需要调用 ID。OpenAI Chat Completions 风格服务则通常返回 `tool_calls[].id` 和 JSON **字符串**形式的 `function.arguments`，回填时必须用对应 `tool_call_id` 配对。接入服务应使用其 SDK 消息结构，不要把本地无 ID 的示例直接当作 OpenAI 协议。

原生协议可以在同一轮返回多个调用；本实验为了突出闭环，主动拒绝多调用。生产实现需要为每个调用独立校验、执行与配对，并管理并发和副作用。

## 四、如何比较与验证

- A/B 都使用模型的 chat template 和相同生成预算；只跑一道题不能证明原生格式“必然更稳定”。
- 观察 `observation.ok`、失败代码与 `status`；打印过请求不代表工具已执行，只有 `execute_once` 的白名单分支会调用函数。
- 可直接用 `execute_once` 测试 `city=3`、缺 city、额外参数、未知 name，预期全部返回 `ok=False`；测试空输出与多个 `<tool_call>`，预期不执行。
- 本例只有固定数据的只读工具，未实现鉴权、超时、并行调用、流式拼接或有界重试。格式合法也不能保证模型选对城市或事实正确。

## 本节小结

两种方式都必须完成校验、单次执行、结构化失败回填和只读收尾。原生模板减轻了格式提示负担，并没有取代服务端验证。下一节进入 **3.1 实现任务分解规划器（Planner）——用 LangGraph 复现同一流程**。
