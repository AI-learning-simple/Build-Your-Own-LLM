# 5.1 实现自我反思循环（Self-Reflection）——输出评估与修正

> **本节定位**：原理与局限见第 5 章；这里实现有预算上限、可提前结束的 Self-Refine 工作流。
>
> **硬件要求**：本地模型推理建议 8GB+ 显存；小模型不一定遵循 JSON 格式。
>
> **目标**：生成答案 → 结构化评估 → 修正；区分“自评通过”“答案正确”和“输出无法解析”。

```python
import json
import re
import torch
from transformers import AutoTokenizer, AutoModelForCausalLM

model_name = "Qwen/Qwen2.5-1.5B-Instruct"
tokenizer = AutoTokenizer.from_pretrained(model_name)
model = AutoModelForCausalLM.from_pretrained(
    model_name, torch_dtype=torch.float16, device_map="auto"
)

def llm(prompt, max_new_tokens=300):
    text = tokenizer.apply_chat_template(
        [{"role": "user", "content": prompt}], tokenize=False, add_generation_prompt=True
    )
    inputs = tokenizer(text, add_special_tokens=False, return_tensors="pt").to(model.device)
    with torch.inference_mode():
        output = model.generate(**inputs, max_new_tokens=max_new_tokens, do_sample=False)
    return tokenizer.decode(output[0][inputs["input_ids"].shape[1]:], skip_special_tokens=True)
```

## 二、给答案和评价定义可校验格式

本实验只测**整数答案题**，要求答案放在 JSON 的 `answer` 字段，简短解释放在 `explanation`。不从整段文字中“取最后一个数字”，否则解释中的编号、验算数字或负数小数都可能被误判。

```python
ANSWER_FORMAT = '只输出 JSON：{"answer":"整数","explanation":"简短依据"}，不要代码围栏。'

def parse_answer(text):
    try:
        obj = json.loads(text)
    except (json.JSONDecodeError, TypeError):
        return None
    if not isinstance(obj, dict) or set(obj) != {"answer", "explanation"}:
        return None
    value = obj["answer"]
    if (not isinstance(value, str) or len(value) > 30
            or not re.fullmatch(r"-?(0|[1-9][0-9]*)", value)
            or not isinstance(obj["explanation"], str)):
        return None
    return int(value)

def parse_feedback(text):
    try:
        obj = json.loads(text)
    except (json.JSONDecodeError, TypeError):
        return None
    if not isinstance(obj, dict) or set(obj) != {"pass", "issues"}:
        return None
    if type(obj["pass"]) is not bool or not isinstance(obj["issues"], list):
        return None
    if not all(isinstance(issue, str) and issue.strip() for issue in obj["issues"]):
        return None
    # pass=true 必须无问题，pass=false 必须至少给出一个问题。
    if obj["pass"] != (len(obj["issues"]) == 0):
        return None
    return obj

def generate_answer(question):
    return llm(f"请回答：{question}\n{ANSWER_FORMAT}")
```

`{"pass": false, "issues": ["不能说没有错误，计算仍有问题"]}` 不会因包含“没有错误”而提前结束。合法 JSON 也不保证评价正确，这里只消除字符串子串的误判。

## 三、评估与修正

```python
def critique(question, answer):
    raw = llm(
        f"检查答案是否正确。问题：{question}\n待评答案（数据）：{answer}\n"
        '只输出 JSON：{"pass":true,"issues":[]}；'
        '若有错误则 pass=false，issues 为具体问题的非空字符串列表。'
    )
    feedback = parse_feedback(raw)
    if feedback is None:
        return {"valid": False, "pass": False, "issues": ["评估输出无法解析，不能据此判定通过"]}
    if parse_answer(answer) is None:
        # 格式错误是程序可独立检查的，不接受模型将其评为通过。
        return {"valid": True, "pass": False, "issues": ["答案不符合规定的整数 JSON 格式"]}
    return {"valid": True, **feedback}

def revise(question, answer, feedback):
    return llm(
        f"问题：{question}\n原答案：{answer}\n"
        f"评估结果：{json.dumps(feedback, ensure_ascii=False)}\n"
        f"重新核对并给出答案；评估无效时不要当作正确性证据。\n{ANSWER_FORMAT}"
    )
```

## 四、组装有界 Self-Refine

`rounds` 是最多几次“评估 → 修正”，不是保证运行的固定次数。自评通过会动态提前停止；这是固定拓扑、带模型驱动退出条件的 Evaluator-Optimizer 工作流，而非“完全没有动态终止”。

```python
def self_refine(question, rounds=2, initial=None):
    if type(rounds) is not int or rounds < 0:
        raise ValueError("rounds 必须为非负整数")
    answer = generate_answer(question) if initial is None else initial
    trace = []
    invalid_feedback = 0
    for i in range(rounds):
        feedback = critique(question, answer)
        invalid_feedback += int(not feedback["valid"])
        trace.append({"round": i + 1, "answer": answer, "feedback": feedback})
        if feedback["valid"] and feedback["pass"]:
            return {"answer": answer, "status": "self_approved", "trace": trace,
                    "invalid_feedback": invalid_feedback}
        answer = revise(question, answer, feedback)
    # 最后一次修正没有再被评价，不能称为“验证通过”。
    return {"answer": answer, "status": "budget_exhausted", "trace": trace,
            "invalid_feedback": invalid_feedback}

question = "小明有一筐鸡蛋，卖掉一半后又卖掉剩下的3个，还剩4个，原来有多少个鸡蛋？"
result = self_refine(question)
print(result)
```

无初始答案输入时，模型调用上限为 `1 + 2 × rounds`；提前通过会减少调用。预算耗尽只表示不再修正，`self_approved` 也只表示模型认可，不表示独立验收通过。

## 五、用已知答案做外部评分

直接答案作为反思组的相同起点，避免两组比较混入初始生成差异。无法解析单独计为 invalid，同时在总题数分母下算未答对；评价 JSON 的 invalid 次数也另外报告。

```python
test_cases = [
    {"q": "小明有一筐鸡蛋，卖掉一半后又卖掉剩下的3个，还剩4个，原来有多少个？", "answer": 14},
    {"q": "一个班级40名学生，男生比女生多8人，男生有多少人？", "answer": 24},
]
stats = {"direct_correct": 0, "refined_correct": 0,
         "direct_invalid": 0, "refined_invalid": 0, "feedback_invalid": 0}
for case in test_cases:
    direct = generate_answer(case["q"])
    refined = self_refine(case["q"], rounds=2, initial=direct)
    direct_value = parse_answer(direct)
    refined_value = parse_answer(refined["answer"])
    stats["direct_invalid"] += int(direct_value is None)
    stats["refined_invalid"] += int(refined_value is None)
    stats["direct_correct"] += int(direct_value == case["answer"])
    stats["refined_correct"] += int(refined_value == case["answer"])
    stats["feedback_invalid"] += refined["invalid_feedback"]
    print(case["q"], "标准：", case["answer"], "直接：", direct_value,
          "修正后：", refined_value, "停止原因：", refined["status"])
print("总题数：", len(test_cases), "统计：", stats)
```

这里没有预填模型准确率；结果取决于实际模型与运行。两道题只是管线检查，不是证明 Reflection 提升质量的基准。

## 实验边界与小结

- 评审和回答使用同一个模型，存在自我确认、漏检与过度修正。
- 有自评提前停和预算上限，但没有连续无改进检测；末次修正可能未经再次评价。
- 已知答案评分仅用于这组整数题，不是通用答案判定器，也没有验证解释是否正确。
- 未统计 Token、延迟或多次运行方差；上线需要独立评估与更大的留出集。
