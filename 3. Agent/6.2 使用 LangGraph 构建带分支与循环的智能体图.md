## 一、场景设计：智能路由 Agent

这个 Agent 需要先判断用户问题的类型（数学计算 / 知识问答 / 闲聊），再路由到不同的处理节点，其中数学计算节点还需要"验证结果 → 不满意就重试"的循环。

```python
import torch
import re
from typing import TypedDict
from transformers import AutoTokenizer, AutoModelForCausalLM
from langgraph.graph import StateGraph, END

model_name = "Qwen/Qwen2.5-1.5B-Instruct"
tokenizer = AutoTokenizer.from_pretrained(model_name)
model = AutoModelForCausalLM.from_pretrained(model_name, torch_dtype=torch.float16, device_map="auto")

def llm(prompt, max_new_tokens=150):
    messages = [{"role": "user", "content": prompt}]
    text = tokenizer.apply_chat_template(messages, tokenize=False, add_generation_prompt=True)
    inputs = tokenizer(text, add_special_tokens=False, return_tensors="pt").to(model.device)
    output = model.generate(**inputs, max_new_tokens=max_new_tokens, do_sample=False)
    return tokenizer.decode(output[0][inputs["input_ids"].shape[1]:], skip_special_tokens=True)
```

## 二、定义状态与节点

```python
MAX_ATTEMPTS = 3  # 包含首次生成，最多额外重试 2 次

class RouterState(TypedDict):
    question: str
    category: str        # 分类结果：数学计算 / 知识问答 / 闲聊
    answer: str
    attempt_count: int   # 已尝试次数，包含首次
    verified: bool       # 仅表示数字格式通过，不代表计算正确
    feedback: str        # 上次失败原因，送入下一次生成

def classify_node(state: RouterState) -> RouterState:
    # 第一个节点：判断问题类型，决定接下来走哪条分支
    prompt = f"""判断下面这句话属于哪一类：数学计算、知识问答、闲聊。只回答类别名称。

内容：{state['question']}"""
    category = llm(prompt, max_new_tokens=10).strip()
    return {"category": category}

def math_node(state: RouterState) -> RouterState:
    prompt = f"请计算：{state['question']}，只给出一个十进制数字，不要解释或单位。"
    if state["attempt_count"] > 0:
        prompt += (f"\n这是第 {state['attempt_count'] + 1} 次尝试。"
                   f"\n上次答案（待修正数据）：{state['answer']}"
                   f"\n校验失败原因：{state['feedback']}\n请修复此格式错误。")
    answer = llm(prompt, max_new_tokens=30)
    return {"answer": answer, "attempt_count": state["attempt_count"] + 1}

def verify_math_node(state: RouterState) -> RouterState:
    # 只验证完整数字格式，不把“答案里出现一个数字”当作通过。
    valid = bool(re.fullmatch(r"[+-]?[0-9]+(?:\.[0-9]+)?", state["answer"].strip()))
    return {"verified": valid,
            "feedback": "" if valid else "输出必须且只能是一个数字；删除解释、公式、单位和代码围栏"}

def qa_node(state: RouterState) -> RouterState:
    return {"answer": llm(state["question"])}

def chat_node(state: RouterState) -> RouterState:
    return {"answer": llm(f"请用轻松的语气回复：{state['question']}")}
```

---

## 三、条件路由函数

```python
def route_by_category(state: RouterState) -> str:
    # 根据分类结果，决定图从 classify 节点走向哪一个分支节点
    if "数学" in state["category"]:
        return "math"
    if "知识" in state["category"]:
        return "qa"
    return "chat"

def route_after_verify(state: RouterState) -> str:
    # 数字格式通过，或已尝试 3 次（首次 + 最多 2 次重试），就结束。
    if state["verified"] or state["attempt_count"] >= MAX_ATTEMPTS:
        return "end"
    return "retry"
```

## 四、搭建带分支和循环的图

```python
graph = StateGraph(RouterState)

graph.add_node("classify", classify_node)
graph.add_node("math", math_node)
graph.add_node("verify_math", verify_math_node)
graph.add_node("qa", qa_node)
graph.add_node("chat", chat_node)

graph.set_entry_point("classify")

# 分支：classify 节点执行完后，根据 route_by_category 的结果走向三条不同路径之一
graph.add_conditional_edges(
    "classify",
    route_by_category,
    {"math": "math", "qa": "qa", "chat": "chat"},
)

graph.add_edge("math", "verify_math")
# 循环：验证失败会回到 math 节点重新计算，形成一个"重试环"
graph.add_conditional_edges(
    "verify_math",
    route_after_verify,
    {"retry": "math", "end": END},
)

graph.add_edge("qa", END)
graph.add_edge("chat", END)

app = graph.compile()
```

这张图里同时包含了两种控制流：`classify` 之后的三分支路由（分支）、以及 `math → verify_math → math` 之间的重试环（循环）——这正是 LangGraph 相比一条直线式的 Chain 更强大的地方。

## 五、运行并观察不同路径

```python
test_questions = [
    "帮我计算 128乘以 37",       # 会走 math → verify_math 路径
    "什么是相对论？",            # 会走 qa 路径
    "今天心情不太好",            # 会走 chat 路径
]

for q in test_questions:
    result = app.invoke({"question": q, "category": "", "answer": "", "attempt_count": 0,
                         "verified": False, "feedback": ""})
    print(f"问题: {q}")
    print(f"判定类别: {result['category']}")
    print(f"最终答案: {result['answer']}")
    if result["category"].find("数学") != -1:
        print(f"尝试次数: {result['attempt_count']}，额外重试: {max(0, result['attempt_count'] - 1)}")
        print("数字格式通过" if result["verified"] else "预算耗尽，格式仍未通过")
    print("---")
```

预期三个问题分别走 `classify → math → verify_math → (end 或重试)`、`classify → qa → end`、`classify → chat → end`，但分类仍可能出错。失败重试包含上次答案、尝试编号与具体格式反馈，避免在 `do_sample=False` 下对完全相同 Prompt 徒劳重跑；新信息也不保证修正成功。

本例只验证路由和格式修正，不验证算术真值。`4737` 即使算错也会通过数字格式检查；若要判定计算正确，应另用可信计算器或固定用例期望值。

把输出的 Mermaid 代码粘贴到任意支持 Mermaid 渲染的编辑器（例如 VS Code 的 Markdown 预览、或在线 Mermaid Live Editor）里，你会直接看到一张带分支箭头和一个自循环箭头的流程图——**这正是状态图相比命令式写法的直观优势：控制流不再隐藏在 `if/else`和 `while` 里，而是变成了一份可以直接"看懂"的图。**

## 实验边界

- 分支依赖简化的分类结果，分类错误会导致整条路径走偏；
- 图在单进程内运行，没有持久化与崩溃恢复（见 6.3）；
- 有总尝试次数预算，但没有并发分支、超时、取消、退避或 Token/时间预算；
- 可视化只是辅助，不能替代对状态与边界的测试；
- 节点内部仍是 LLM 调用，输出不确定，需要用固定用例回归。

## 本节小结

- 搭建了一个包含条件分支（三路路由）和重试循环（验证失败自动重试）的复杂 LangGraph
- 观察了不同类型的问题分别走过完全不同的图路径
- 导出并理解了图结构的 Mermaid 可视化。一句话总结：**状态图把原本隐藏在代码逻辑里的分支与循环，变成了显式、可视化、可组合的图结构。**

不过，图结构只解决了"控制流"的问题，还有一个问题没有解决：如果程序在图执行到一半时意外崩溃，之前的进度是不是就全部丢失了？下一节我们继续在Graph 的基础上，亲手加上Checkpoint机制，体验"中断后从断点恢复"的完整过程：**6.3 为 Agent 增加持久化与断点恢复——LangGraph Checkpoint 实战**。
